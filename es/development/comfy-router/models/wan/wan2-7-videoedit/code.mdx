type Result = { output: { video_url: string } };

      },
    ],
  },
  parameters: {
    resolution: "720P",
    audio_setting: "origin",
  },
});
console.log("requestId:", handle.requestId); // junto con el ID del modelo, es todo lo que necesita otro proceso

// Sondea hasta que la solicitud se completa, esperando el Retry-After que indica el servidor.
for await (const update of handle.events()) {

## Ejemplos

### Entrada

```json
{
  "input": {
    "prompt": "restyle the clip as a hand-drawn ink sketch, keep the dancer's movement unchanged",
    "media": [
      {
        "type": "video",
        "url": "https://raw.githubusercontent.com/Comfy-Org/workflow_templates/main/input/street_dancer.mp4"
      }
    ]
  },
  "parameters": {
    "resolution": "720P",
    "audio_setting": "origin"
  }
}
```
