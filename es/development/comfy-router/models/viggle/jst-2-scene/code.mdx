El mismo cuerpo, enviado a `POST https://api.comfy.org/v2/models/viggle/jst-2-scene/requests`. Router responde `201` con un `request_id` en cuanto se admite la ejecución, y puedes recoger el resultado una vez que está listo, desde este proceso o desde otro. [Entrega en cola](/es/development/comfy-router/queue) explica el estado, la cancelación y la recogida.

```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Lee COMFY_API_KEY del entorno.
# Cada llamada a submit() genera su propia Idempotency-Key y la reutiliza en los reintentos automáticos.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "viggle/jst-2-scene",
            {
                "prompt": "a sunlit atrium with tall plants and a stone floor",
                "quality": "low",
                "sessionId": "replace-with-a-new-random-id",
            },
        )
        print("request_id:", handle.request_id)  # junto con el ID del modelo, es todo lo que necesita otro proceso

        # Consulta hasta que la solicitud se complete, esperando el Retry-After que indica el servidor.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # El payload propio del proveedor, el mismo valor que devuelve models.run().
        # Una solicitud que falló o se canceló genera aquí el error tipado de Router.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Lee COMFY_API_KEY del entorno.
// Cada llamada a submit() genera su propia Idempotency-Key y la reutiliza en los reintentos automáticos.
const handle = await comfy.models.submit("viggle/jst-2-scene", {
  prompt: "a sunlit atrium with tall plants and a stone floor",
  quality: "low",
  sessionId: "replace-with-a-new-random-id",
});
console.log("requestId:", handle.requestId); // junto con el ID del modelo, es todo lo que necesita otro proceso

// Consulta hasta que la solicitud se complete, esperando el Retry-After que indica el servidor.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// El mismo resultado que devuelve models.run(). Una solicitud que falló o se canceló se rechaza aquí.
const result = await handle.get();

console.log(result.data);
```
