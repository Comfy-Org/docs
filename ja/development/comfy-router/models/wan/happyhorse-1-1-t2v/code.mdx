import asyncio
from comfy_sdk import AsyncComfy

# COMFY_API_KEY を環境変数から読み取ります。
# 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライのためにそれを再利用します。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "wan/happyhorse-1.1-t2v",
            {
                "input": {
                    "prompt": "a single red maple leaf falling onto still water, slow motion, shallow depth of field",
                },
                "parameters": {
                    "resolution": "720P",
                    "duration": 5,
                },
            },
        )
        print("request_id:", handle.request_id)  # モデル ID と合わせて、別のプロセスに必要なのはこれだけです

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の秒数だけ待機します。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # プロバイダー自身のペイロードで、models.run() が返す値と同じです。
        # 失敗またはキャンセル済みのリクエストは、ここで型付きの Router エラーを発生させます。

    print("video:", result["output"]["video_url"])

<ParamField body="input.img_url" type="string">

<ParamField body="input.media" type="object[]">
  メディアアセットのリスト。動画生成のための参照素材（画像、オーディオ、ビデオ）を指定します。
各要素は type と url フィールドを持ちます。このフィールドはこのコンポーネントが提供するすべての
モデルで共有されており、あるモデルがどの type 値を受け付けるか、また画像やビデオの入力を
img_url や reference_video_urls ではなくここで持つかどうかは、モデルによって異なります。
モデルごとの対応 type 値:
- wan2.7-i2v: first_frame、last_frame、driving_audio、first_clip。first_frame と
  last_frame: JPEG/JPG/PNG/BMP/WEBP、1 辺 240-8000 ピクセル、アスペクト比 1:8 から 8:1 の間、
  最大 20MB、パブリック URL または data:\{MIME_type};base64,... URL。
  driving_audio（WAV/MP3、2-30 秒、最大 15MB）と first_clip（MP4/MOV、2-10 秒、
  1 辺 240-4096 ピクセル、最大 100MB）: パブリックな HTTP/HTTPS URL。インラインの
  data: URL は受け付けられません。
- wan2.7-r2v: first_frame（任意、最大 1）、reference_image、reference_video。少なくとも
  1 つの reference_image または reference_video が必要で、reference 画像と
  reference ビデオの合計は最大 5 です。first_frame と reference_image: JPEG/JPG/PNG/BMP/WEBP
  （アルファチャンネルなし）、1 辺 240-8000 ピクセル、アスペクト比 1:8 から 8:1 の間、最大
  20MB、パブリック URL または data:\{MIME_type};base64,... URL。reference_video: パブリックな
  HTTP/HTTPS URL。インラインの data: URL は受け付けられません。
- wan2.7-videoedit: video（正確に 1 つ。編集対象のクリップ: MP4/MOV、2-10 秒、
  1 辺 240-4096 ピクセル、最大 100MB、パブリックな HTTP/HTTPS URL。インラインの data: URL
  は受け付けられません）に加えて、任意で 0 から 4 つの reference_image（JPEG/JPG/PNG/BMP/WEBP、
  1 辺 240-8000 ピクセル、アスペクト比 1:8 から 8:1 の間、最大 20MB、パブリック URL
  または data:\{MIME_type};base64,... URL）。
- wan3.0-video、wan3.0-video-prime: first_frame（最大 1）、last_frame（最大 1）、
  reference_image（最大 10）、reference_video（最大 5 クリップ、再生時間の合計 \<= 15 秒）、
  reference_audio（最大 5 クリップ、再生時間の合計 \<= 15 秒）、file（最大 1、link とは併用不可）、
  link（最大 1、file とは併用不可）。reference_*/file/link の type と
  first_frame/last_frame の type は、同じリクエスト内では同時に使用できません。
  配列の順序が、プロンプト内でのアセットの参照順（Image 1、Video 1、Audio 1、...）を定義します。
  first_frame、last_frame、reference_image: JPEG/JPG/PNG/BMP/WEBP、1 辺 240-8000 ピクセル、
  アスペクト比 最大 8:1、最大 20MB、パブリック URL または
  data:\{MIME_type};base64,... URL。reference_video、reference_audio、file、link: パブリックな
  HTTP/HTTPS URL。インラインの data: URL は受け付けられません。
- happyhorse-1.0-i2v、happyhorse-1.1-i2v: first_frame のみ、正確に 1 つ。これらの表記は
  最初のフレームをここで受け取り、img_url では受け取りません。最低 300x300 ピクセル、アスペクト比
  1:2.5 から 2.5:1 の間、JPEG/JPG/PNG/WEBP、最大 20MB、パブリック URL または
  data:\{MIME_type};base64,... URL。
- happyhorse-1.0-r2v、happyhorse-1.1-r2v: reference_image のみ、1 から 9 つ。これが
  このモデル唯一の参照入力です。最短辺が最低 400 ピクセル、JPEG/JPG/PNG/WEBP、
  最大 20MB、パブリック URL または data:\{MIME_type};base64,... URL。reference_video は
  この操作の入力 type ではありません。
- happyhorse-1.0-video-edit: video（正確に 1 つ。編集対象のクリップ: MP4/MOV、3-60 秒、
  長辺が最大 4096 ピクセル、短辺が最低 360 ピクセル、アスペクト比 1:2.5 から 2.5:1 の間、
  最大 100MB、パブリックな HTTP/HTTPS URL。インラインの data: URL は受け付けられません）に
  加えて、任意で 0 から 5 つの reference_image（JPEG/JPG/PNG/WEBP、最低 300x300 ピクセル、
  アスペクト比 1:2.5 から 2.5:1 の間、最大 20MB、パブリック URL または
  data:\{MIME_type};base64,... URL）。
- happyhorse-1.0-t2v、happyhorse-1.1-t2v、wan2.7-t2v、wan2.6-t2v、wan2.5-t2v-preview:
  テキストから動画への操作。これらのパートナー向けリファレンスにはメディア入力が記載されていないため、
  このフィールドは未設定のままにしてください。
- wan2.5-i2v-preview、wan2.6-i2v、wan2.6-r2v: このコンポーネントはこれらの表記向けのメディア
  ボキャブラリを列挙していません。wan2.5-i2v-preview と wan2.6-i2v は最初のフレームを
  img_url で受け取り、wan2.6-r2v は参照クリップを reference_video_urls で受け取ります。
上記のモデルごとの個数はパートナーの制約であり、このスキーマの制約ではありません。media には
minItems/maxItems が宣言されていません。許容される type とその上限がモデルごとに異なるためです。
スキーマが受け付けることはプロバイダーが受け付けることを意味しません。また逆方向について言えば、
上記のアセットごとの「最大 20MB」という数値（画像入力の上限であり、インラインで送信できる唯一の入力）
は、最終的にパートナーが受け取るアセットに対するパートナーの上限であり、アセットがパブリック URL の
場合は記載どおりに適用されます。そのバイト列は Comfy を経由しないためです。インラインの
data:\{MIME_type};base64,... URL は Comfy を経由するため、転送時の上限にも直面します。
Comfy Router の POST ボディは合計 100 MiB に制限され、それを超えると 413 で応答されます。
また base64 はペイロードを約 4/3 に膨らませるため、およそ 75 MB を超える単一のインラインアセットは、
ここに記載されたパートナーのルールに到達する前に拒否されます。そのサイズでは、パートナー自身の20MB の上限はインラインで余裕をもって収まります。そのため、1 つのアセットについてはパートナーのルールが制約となります。一方、トランスポートの上限は複数のアセットにまたがって制約します。100 MiB の Router の上限は各要素ではなくリクエスト全体を制限するためです。これらの上限に近いものはパブリック URL として送信してください。

<ParamField body="input.prompt" type="string">
  テキストプロンプト。中国語と英語をサポートします。文書化された長さを超えると、wan2.x モデルはプロンプトを拒否するのではなく切り詰めます。wan2.5 と wan2.6 では 1,500 文字、wan2.7 では 5,000 文字です。wan3.0-video と wan3.0-video-prime は最大 20,000 文字を受け付けます。プロバイダーは、prompt と negative_prompt を合わせて約 60,850 バイトの UTF-8 テキスト (約 20,000 文字の中国語) を超えるリクエストを拒否し、InvalidParameter `Range of input length should be [1, 61440]` を返します。
  複数の参照ビデオを使う wan2.6-r2v では、'character1'、'character2' などを使って、参照ビデオの順に被写体を指定します。例: "Character1 sings on the roadside, Character2 dances beside it"
  wan3.0-video の参照モードでは、'Image 1'、'Video 1'、'Audio 1' などを使って、media 配列内の対応する順序でメディアアセットを指定します。

<ParamField body="input.reference_video_urls" type="string[]">
  wan2.6-r2v モデル専用の参照ビデオ URL です。1～3 本のビデオ URL の配列。
入力の制限:
- 形式: mp4, mov
- 本数: 1～3 本のビデオ
- 1 本のビデオの長さ: 2～30 秒
- 単一ファイルサイズ: 最大 30MB
- audio_url と併用できません
参照の再生時間: ビデオ 1 本なら最大 5 秒、2 本ならそれぞれ最大 2.5 秒、3 本なら比例して短くなります。
課金: 実際に使用された参照の再生時間に基づきます。
</ParamField>

- wan3.0-video: ビデオ入力なしの場合は [2, 30] の整数。ビデオ入力ありの場合は、入力ビデオの再生時間と出力ビデオの再生時間の合計が 30 秒を超えてはなりません。-1 を指定すると、モデルが適切な再生時間を選択するスマート再生時間モードが有効になります
このプロキシは [-1, 30] の範囲外の再生時間に対して 400 を返します。

  範囲: `-1` から `30`
</ParamField>

<ParamField body="parameters.prompt_extend" type="boolean" default="true">
  プロンプトのインテリジェントな書き換えを有効にするかどうか。デフォルトは true です
</ParamField>

<ParamField body="parameters.ratio" type="string">
  生成されるビデオのアスペクト比。wan2.7 および wan3.0 モデル専用です。
wan2.7 モデルでは、指定しない場合、解像度ティアに基づいてデフォルトが決まります。
wan3.0-video では、adaptive (デフォルト) が適切なものを自動的に推奨します
</ParamField>

入力メディアの比率と意図に基づくアスペクト比。

  指定可能な値: `adaptive`、`16:9`、`9:16`、`1:1`、`4:3`、`3:4`
<ParamField body="parameters.resolution" type="string">
  解像度レベル。サポートされる値はモデルによって異なります:
- wan2.5-i2v-preview: 480P、720P、1080P
- wan2.6 モデル (t2v、i2v、r2v): 720P、1080P のみ (480P のサポートなし)
- wan2.7 モデル (i2v、t2v、r2v、videoedit): 720P、1080P (デフォルト 1080P)
- wan3.0-video、wan3.0-video-prime: 480P、720P、1080P (上流のデフォルト 1080P)
- happyhorse-1.0-* (t2v、i2v、r2v、video-edit): 720P、1080P
- happyhorse-1.1-* (t2v、i2v、r2v): 480P、720P、1080P
このプロキシは、resolution も size も指定されていないビデオ生成リクエストを拒否します。課金レートを選択するのは解像度ティアだからです。resolution をチェックする前に大文字に変換し、大文字に変換した値をそのまま転送します。この enum に含まれない値に対しては 400 を返し、上記で 480P がないとされている wan2.6、wan2.7、happyhorse-1.0 の各モデルでは、480P に対して 400 を返します (resolution として送信された場合も、480P の size として送信された場合も同様です)。

  指定可能な値: `480P`、`720P`、`1080P`
</ParamField>

<ParamField body="parameters.seed" type="integer">
  乱数シード。モデルが生成するコンテンツのランダム性を制御するために使用します。

  範囲: `0` から `2147483647`
</ParamField>

<ParamField body="parameters.shot_type" type="string" default="&quot;single&quot;">
  インテリジェントなマルチレンズ制御。prompt_extend が有効な場合にのみ動作します。
wan2.6 および wan2.7-r2v モデル向け。
- single: 単一ショットのビデオ (デフォルト)
- multi: マルチショットのビデオ

  指定可能な値: `multi`、`single`
</ParamField>

<ParamField body="parameters.size" type="string">
  幅*高さ形式のビデオ解像度。サポートされる解像度はモデルによって異なります:
wan2.5 T2V の場合: 480P (480*832、832*480、624*624)、720P、1080P の各サイズ
wan2.6 T2V/R2V の場合 (480P なし):
  720P: 1280*720、720*1280、960*960、1088*832、832*1088
  1080P: 1920*1080、1080*1920、1440*1440、1632*1248、1248*1632
size と resolution の両方が送信された場合、課金ティアを選択するのは size であり、resolution はそのモデルに対してこのプロキシが受け付ける値である必要があります。480P の size は、解像度の説明で 480P がないとされているすべてのモデルで 400 を返します。
</ParamField>

<ParamField body="parameters.watermark" type="boolean" default="false">
  ウォーターマークロゴを追加するかどうか。ウォーターマークは右下隅に配置されます。
</ParamField>
