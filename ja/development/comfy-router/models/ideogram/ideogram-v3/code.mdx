from comfy_sdk import Comfy

# 環境変数から COMFY_API_KEY を読み取ります。
# 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライのために再利用します。
with Comfy() as client:
    handle = client.models.submit(
        "ideogram/ideogram-v3",

            "rendering_speed": "TURBO",

    print("request_id:", handle.request_id)  # モデル ID と合わせて、別のプロセスが必要とするすべて

    # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After だけ待機します。

    # プロバイダー自身のペイロードで、models.run() が返すのと同じ値です。
