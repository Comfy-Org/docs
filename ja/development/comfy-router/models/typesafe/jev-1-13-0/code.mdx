import asyncio
from comfy_sdk import AsyncComfy

# 環境変数から COMFY_API_KEY を読み取ります。

async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "typesafe/jev-1.13.0",
            {
                "questions": {
                    "department": {
                        "criteria": {
                            "billing": "Payments, invoicing, refunds",

                        "type": "noul",
                    },
                },
                "state": "Help! My payouts have been failing for 3 days.",
            },
        )
        print("request_id:", handle.request_id)  # モデル ID と合わせて、別プロセスに必要な情報はこれだけです

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の時間だけ待機します。

    },
    is_urgent: {
      instructions: "Does this convey urgency?",
      type: "noul",
    },

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の時間だけ待機します。
for await (const update of handle.events()) {

## スキーマ

### 入力

<ParamField body="assets" type="object">
  リクエスト内の画像。参照文字列をキーとします。すべての参照は `state`、`instructions`、`criteria` のいずれかに現れる必要があります。各画像は一辺 280～1568 ピクセル、デコード後 16 MiB 以下の静止 PNG または JPEG です。TypeSafe は合計 32 MiB を超える呼び出しを拒否します。
</ParamField>

<ParamField body="model" type="string">
  ピン留めされた Jev バージョン。任意: Router はパス内のモデル ID からこれを設定します。`jev-latest` と `jev-preview` のエイリアスは受け付けられません。
</ParamField>

<ParamField body="questions" type="object" required>
  回答する質問。任意の ID をキーとして指定します。各質問への回答は同じ ID の下に返されます。
</ParamField>

<ParamField body="state" type="object" required>
  評価するコンテンツ: 文字列、または最大 64 階層までの構造化データ (オブジェクトまたは配列)。その中のどこかにあるアセット参照は、その画像に置き換えられます。
</ParamField>
