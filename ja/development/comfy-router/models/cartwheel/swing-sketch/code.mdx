import asyncio
from comfy_sdk import AsyncComfy

# 環境から COMFY_API_KEY を読み取ります。
# 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動再試行のために再利用します。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "cartwheel/swing-sketch",
            {
                "exportSettings": {
                    "characterID": "char-upload-MHR",
                    "exportType": "glb",
                    "forward": "-Z",
                    "frameRate": 30,

                    "up": "Y",
                },
                "prompts": ["a person waves with the right hand"],
                "swingSketchProperties": {
                    "duration": 4,
                },
            },
        )
        print("request_id:", handle.request_id)  # モデル ID とともに、別のプロセスが必要とするすべての情報

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After を待ちます。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # プロバイダー自身のペイロード。models.run() が返すのと同じ値です。

        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// 環境から COMFY_API_KEY を読み取ります。
// 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動再試行のために再利用します。
const handle = await comfy.models.submit("cartwheel/swing-sketch", {
  exportSettings: {
    characterID: "char-upload-MHR",
    exportType: "glb",
    forward: "-Z",
    frameRate: 30,
    frameStepSize: 1,
    up: "Y",
  },
  prompts: ["a person waves with the right hand"],
  swingSketchProperties: {
    duration: 4,
  },
});
console.log("requestId:", handle.requestId); // モデル ID とともに、別のプロセスが必要とするすべての情報

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After を待ちます。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返すのと同じ結果です。失敗した、またはキャンセルされたリクエストは、ここで reject されます。
