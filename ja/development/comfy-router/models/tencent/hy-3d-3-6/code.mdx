import asyncio
from comfy_sdk import AsyncComfy

# 環境から COMFY_API_KEY を読み取ります。

async def main():

        handle = await client.models.submit(

            {
                "generate_type": "Geometry",
                "image_url": "https://example.invalid/tencent/hy-3d-3.6/input.png",
            },

        print("request_id:", handle.request_id)  # モデル ID と合わせて、別のプロセスに必要な情報はこれだけです

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After を待ちます。
        async for update in handle.iter_events():

        # プロバイダー自身のペイロードであり、models.run() が返すのと同じ値です。
        # 失敗またはキャンセル済みのリクエストは、ここで型付きの Router エラーを発生させます。

import { comfy } from "@comfyorg/sdk";

// 環境から COMFY_API_KEY を読み取ります。
// 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動再試行のために再利用します。
const handle = await comfy.models.submit("tencent/hy-3d-3.6", {
  generate_type: "Geometry",
  image_url: "https://example.invalid/tencent/hy-3d-3.6/input.png",

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After を待ちます。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返すのと同じ結果です。失敗またはキャンセル済みのリクエストは、ここで拒否されます。
const result = await handle.get();

console.log(result.data);
