import asyncio
from comfy_sdk import AsyncComfy

# 環境変数から COMFY_API_KEY を読み取ります。

async def main():

        handle = await client.models.submit(

                "aspect_ratio": "16:9",
                "prompt": "a lighthouse on a cliff at sunset, oil painting",
                "seed": 12345,

        print("request_id:", handle.request_id)  # モデル ID と合わせて、別のプロセスに必要な情報はこれだけです

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After だけ待機します。
        async for update in handle.iter_events():

        # プロバイダー自身のペイロードで、models.run() が返す値と同じです。

import { comfy } from "@comfyorg/sdk";

// 環境変数から COMFY_API_KEY を読み取ります。
// submit() を呼び出すたびに独自の Idempotency-Key を発行し、自動リトライ時に再利用します。
const handle = await comfy.models.submit("stability/sd3.5-flash", {
  aspect_ratio: "16:9",
  prompt: "a lighthouse on a cliff at sunset, oil painting",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // モデル ID と合わせて、別のプロセスに必要な情報はこれだけです

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After だけ待機します。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返すのと同じ結果です。失敗またはキャンセルされたリクエストはここで reject されます。
const result = await handle.get();

console.log(result.data);

## スキーマ

### 入力

<ParamField body="aspect_ratio" type="string">
  テキストから画像へのみ。約1メガピクセルでの出力アスペクト比。デフォルトは 1:1。

  指定可能な値: `1:1`、`16:9`、`9:16`、`21:9`、`9:21`、`3:2`、`2:3`、`5:4`、`4:5`
</ParamField>

<ParamField body="cfg_scale" type="number">
  画像がプロンプトにどれだけ厳密に従うか。

  範囲: `1` から `10`
</ParamField>

<ParamField body="image" type="string (uri)">
  画像から画像へのみ。ソース画像の Comfy 署名付きアセットURL。

  形式: `uri`
</ParamField>

<ParamField body="mode" type="string">
  デフォルトはテキストから画像へ。画像から画像へでは `image` と `strength` が必要です。

  指定可能な値: `text-to-image`、`image-to-image`
</ParamField>

<ParamField body="negative_prompt" type="string">
  画像に含めないもの。
</ParamField>

<ParamField body="output_format" type="string">
  返されるのは `png` のみです。

  指定可能な値: `png`
</ParamField>

<ParamField body="prompt" type="string" required>
  生成する内容。
</ParamField>

<ParamField body="seed" type="integer">
  0 または省略するとランダムなシードが選ばれます。

  範囲: `0` から `4294967294`
</ParamField>

<ParamField body="strength" type="number">
  画像から画像へのみ。ソース画像からどれだけ離れるか。1 はソース画像を無視します。

  範囲: `0` から `1`
</ParamField>

<ParamField body="style_preset" type="string">
  Stability のスタイルプリセット。例えば `photographic`。
</ParamField>
