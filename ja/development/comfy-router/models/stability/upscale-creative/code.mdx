import asyncio
from comfy_sdk import AsyncComfy

# 環境変数から COMFY_API_KEY を読み取ります。
# 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライで再利用します。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "stability/upscale-creative",
            {
                "creativity": 0.3,
                "image": "https://storage.googleapis.com/example-bucket/source.png",
                "prompt": "a detailed photo of a cat",
                "seed": 12345,
            },
        )
        print("request_id:", handle.request_id)  # モデル ID と合わせて、別のプロセスが必要とするすべての情報

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の間待機します。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # プロバイダー自身のペイロードで、models.run() が返すのと同じ値です。
        # 失敗またはキャンセルされたリクエストは、ここで型付きの Router エラーを発生させます。
        result = await handle.get()

    print(result)

asyncio.run(main())
```

import { comfy } from "@comfyorg/sdk";

// 環境変数から COMFY_API_KEY を読み取ります。
// 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライで再利用します。
const handle = await comfy.models.submit("stability/upscale-creative", {
  creativity: 0.3,
  image: "https://storage.googleapis.com/example-bucket/source.png",
  prompt: "a detailed photo of a cat",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // モデル ID と合わせて、別のプロセスが必要とするすべての情報

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の間待機します。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返すのと同じ結果です。失敗またはキャンセルされたリクエストはここで reject されます。
const result = await handle.get();

console.log(result.data);
```

## スキーマ

### 入力

<ParamField body="creativity" type="number">
  どの程度のディテールを創作するか。デフォルト 0.3。

  範囲: `0.1` から `0.5`
</ParamField>

<ParamField body="image" type="string (uri)" required>
  ソース画像の Comfy 署名付きアセット URL。

  形式: `uri`
</ParamField>

<ParamField body="negative_prompt" type="string">
  アップスケール後の画像に含めないもの。
</ParamField>

<ParamField body="output_format" type="string">
  `png` のみ返されます。

  可能な値: `png`
</ParamField>

<ParamField body="prompt" type="string" required>
  アップスケール後の画像に写る内容。
</ParamField>

<ParamField body="seed" type="integer">
  0 または省略時はランダムなシードが選ばれます。

  範囲: `0` から `4294967294`
</ParamField>

<ParamField body="style_preset" type="string">
  `prompt` と `negative_prompt` に適用されるスタイル。

  可能な値: `3d-model`、`analog-film`、`anime`、`cinematic`、`comic-book`、`digital-art`、`enhance`、`fantasy-art`、`isometric`、`line-art`、`low-poly`、`modeling-compound`、`neon-punk`、`origami`、`photographic`、`pixel-art`、`tile-texture`
</ParamField>

Router が `GET /v2/models/stability/upscale-creative/openapi.json` で提供するスキーマから生成されています。これは、リクエストがプロバイダーに到達する前に Router が呼び出しを検証する際に使用する同じドキュメントです。

### 出力

<ResponseField name="finish_reason" type="string" required>
  200 の場合は常に SUCCESS。
</ResponseField>

<ResponseField name="image" type="string (uri)" required>
  （最初の）PNG 画像の Comfy 署名付き URL。最大 24 時間有効。

  形式: `uri`
</ResponseField>

<ResponseField name="images" type="string (uri)[]">
  SDXL のみ。複数の画像が返された場合に使用します。すべての Comfy 署名付き画像 URL を出力順に並べたもので、それぞれ最大 24 時間有効です。
</ResponseField>

<ResponseField name="seed" type="integer" required>
  画像の生成に使用されたシード。
</ResponseField>
