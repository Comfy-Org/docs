```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// 環境から COMFY_API_KEY を読み取ります。
// 各 submit() 呼び出しは独自の Idempotency-Key を発行し、自動リトライのために再利用します。
const handle = await comfy.models.submit("meshy/relief", {
  image_url: "https://example.invalid/meshy/relief/logo.png",
  variant: "low",
});
console.log("requestId:", handle.requestId); // モデル ID と合わせて、別のプロセスに必要なものはこれだけです

// リクエストが完了するまでポーリングし、サーバーが指定した Retry-After の時間だけ待機します。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返すのと同じ結果です。失敗またはキャンセルされたリクエストはここで reject されます。
const result = await handle.get();

console.log(result.data);
```

## スキーマ

### 入力

<ParamField body="base_thickness_mm" type="number">
  ベースプレートの太さ（ミリメートル）。0.1～10。remove_base が true の場合は省略するか 0 を送信します。

  範囲: `0` ～ `10`
</ParamField>

<ParamField body="bend_angle" type="number" default="60">
  曲げ角度（度）。1 未満ではレリーフは平坦なままです。

  範囲: `0` ～ `360`
</ParamField>

<ParamField body="bend_axis" type="string">
  `xz`（デフォルト、左右の端を曲げる）または `yz`（上下の端を曲げる）。bend_mode が convex または concave の場合にのみ適用されます。
</ParamField>

<ParamField body="bend_mode" type="string">
  `off`（デフォルト）、`convex` または `concave`。
</ParamField>

<ParamField body="carving_depth_mm" type="number">
  彫刻の深さ（ミリメートル）。デフォルトは variant が high のとき 6、low のとき 3 です。

  範囲: `0` ～ `60`
</ParamField>

<ParamField body="carving_mode" type="string">
  `emboss`（デフォルト）または `engrave`。
</ParamField>

<ParamField body="color_mapping" type="boolean" default="false">
  画像からモデルに色を付けます。GLB、OBJ、FBX、USDZ は色を保持し、STL と 3MF は白のままです。コストは変わりません。
</ParamField>

<ParamField body="image_url" type="string" required>
  jpg、jpeg、png、webp 画像の公開 URL またはデータ URI。最大 20 MB、最小 32 x 32 ピクセル。透明な領域は黒で埋められます。
</ParamField>

<ParamField body="mesh_resolution" type="integer" default="1024">
  長辺方向の点数。値を大きくするとファイルが大幅に大きくなります（2048 で約 150 MB の GLB、300 MB の STL）。

  範囲: `512` ～ `2048`
</ParamField>

<ParamField body="removal_threshold" type="number" default="0.01">
  背景除去のしきい値。remove_base が true の場合にのみ適用されます。

  範囲: `0.01` ～ `1`
</ParamField>

<ParamField body="remove_base" type="boolean" default="false">
  背景を切り取って、レリーフにベースプレートがない状態にします。
</ParamField>

<ParamField body="should_generate_model" type="boolean" default="true">
  false にすると深度マップのみを生成し、モデルとプレビューは生成されません。コストは変わりません。
</ParamField>

<ParamField body="size_mm" type="number" default="60">
  長辺の長さ（ミリメートル）。

  範囲: `20` ～ `250`
</ParamField>

<ParamField body="target_formats" type="string[]">
  生成するモデルのフォーマット。glb、obj、fbx、stl、usdz、3mf のうち 1 つ以上。デフォルトは glb と stl です。
</ParamField>

<ParamField body="variant" type="string">
  `low`（デフォルト）は浅い深度とより細かいディテールで、レーザーや CNC 加工向けです。`high` はより深い彫刻的なレリーフで、3D プリント向けです。carving_depth_mm のデフォルトも設定します。
</ParamField>

`GET /v2/models/meshy/relief/openapi.json` で Router が提供するスキーマから生成されています。これは、リクエストがプロバイダーに届く前に呼び出しを検証する際に使用されるものと同じドキュメントです。
