from comfy_sdk import Comfy

# Reads COMFY_API_KEY from the environment.
# Each submit() call mints its own Idempotency-Key and reuses it for automatic retries.
with Comfy() as client:
    handle = client.models.submit(
        "openai/gpt-image-1",
        {
            "image": ["https://storage.googleapis.com/comfy-cloud-assets/0199b3f4-1d2e-7a3b-8c4d-5e6f70819234.png"],
            "prompt": "give the rocketship rainbow coloring",
            "quality": "low",
            "size": "1024x1024",
        },
    )
    print("request_id:", handle.request_id)  # with the model ID, all another process needs

    # Poll until the request completes, waiting the Retry-After the server names.
    for update in handle.iter_events():
        print(update.status, update.queue_position)

    # The provider's own payload, the same value models.run() returns.
    # A request that failed or was cancelled raises the typed Router error here.

### 入力

<ParamField body="background" type="string">
  背景の透明度

  可能な値: `transparent`、`opaque`
</ParamField>

<ParamField body="image" type="string | string[]">
  編集するソース画像。このフィールドの存在が編集操作を選択します。テキストから画像への生成では完全に省略してください。
  1 枚の画像またはその配列を受け付けます。gpt-image-1 以降は最大 16 枚までで、Router はディスパッチ前にこのスキーマからそれを強制します。
</ParamField>

<ParamField body="mask" type="string">
  1 枚の画像。呼び出し元の代わりに Router が取得する https URL か、バイト列をインラインで含む `data:image/<format>;base64,<payload>` URI のいずれかの形式で指定します。受け付ける画像形式は png、webp、jpeg で、このセットはプロバイダーに委ねられず解決時に強制されます。svg、gif、avif、tiff はここで拒否され、送信した画像を名指しで示します。原因から 2 ホップ離れたパートナーエラーにはなりません。
  URL 形式を推奨します。base64 ペイロードは約 4/3 に膨らみ、リクエストボディ全体は 10 MiB に制限されるため、インライン形式ではソース画像が約 7.5 MB で上限に達します。URL ならこれを完全に回避できます。
  URL はあなたの認証情報なしで解決できなければなりません。Router はサーバー側で取得し、リクエストに呼び出し元の認証情報を付けません。したがって、署名がクエリ文字列にある Comfy 署名済みアセット URL がサポートされる形式です。`/api/assets/{id}/content` のような認証済みエンドポイントは、画像ではなく 401 を Router に返します。その URL はご自身でリクエストし、リダイレクト先の署名済み URL を渡してください。取得はオープンなインターネットではなく Comfy 自身のアセットバケットに限定され、送信する URL だけでなくすべてのリダイレクトホップにも適用されるため、サードパーティの URL やリダイレクトする URL は拒否されます。
  以下の例は形状のみを示しています。実際の値には署名用のクエリパラメータも含まれますが、ここでは意図的に省略しています。公開された例はスペックに永久に残り、実際の署名は漏洩した認証情報であると同時に機能しなくなるリンクでもあるためです。
  各画像は 25 MiB、1 リクエストの画像は合計 64 MiB が上限です。
</ParamField>
