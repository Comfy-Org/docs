from comfy_sdk import Comfy

# Reads COMFY_API_KEY from the environment.
# Each submit() call mints its own Idempotency-Key and reuses it for automatic retries.
with Comfy() as client:
    handle = client.models.submit(
        "openai/gpt-image-2.5-flare",
        {

            "prompt": "give the rocketship rainbow coloring",

            "size": "1024x1024",

    )
    print("request_id:", handle.request_id)  # with the model ID, all another process needs

    # Poll until the request completes, waiting the Retry-After the server names.
    for update in handle.iter_events():
        print(update.status, update.queue_position)

    # The provider's own payload, the same value models.run() returns.
    # A request that failed or was cancelled raises the typed Router error here.

## スキーマ

### 入力

<ParamField body="background" type="string">
  背景の透明度

  指定可能な値: `transparent`、`opaque`
</ParamField>

<ParamField body="image" type="string | string[]">
  編集するソース画像。このフィールドが存在すると編集操作が選択されます。テキストから画像への生成では完全に省略してください。
1 枚の画像またはその配列を受け付けます。gpt-image-1 以降は最大 16 枚で、Router はディスパッチ前にこのスキーマに基づいてそれを強制します。
</ParamField>

<ParamField body="mask" type="string">
  1 枚の画像を、呼び出し元に代わって Router が取得する https URL、またはバイト列をインラインで含む `data:image/<format>;base64,<payload>` URI のいずれかの形式で指定します。受け付ける画像形式は png、webp、jpeg で、このセットはプロバイダーに任されず解決時に強制されます。svg、gif、avif、tiff はここで拒否され、送信した画像を名指しします。原因から 2 ホップ離れたパートナーエラーになることはありません。
URL 形式を推奨します。base64 ペイロードは約 4/3 に膨張し、リクエストボディ全体は 10 MiB に制限されるため、インライン形式ではソース画像が約 7.5 MB あたりで上限に達します。URL ならこれを完全に回避できます。
URL はあなたの認証情報なしで解決できる必要があります。Router はサーバー側でこれを取得し、リクエストに呼び出し元の認証情報を一切付けません。そのため、クエリ文字列に署名が含まれる Comfy 署名付きアセット URL がサポートされる形式です。`/api/assets/{id}/content` のような認証済みエンドポイントは、画像ではなく 401 を Router に返します。その URL は自分でリクエストし、リダイレクト先の署名付き URL を渡してください。取得は、送信した URL だけでなくすべてのリダイレクトホップにおいて、オープンなインターネットではなく Comfy 自身のアセットバケットに限定されます。そのため、サードパーティの URL やリダイレクトする URL は拒否されます。
以下の例は形状のみを示しています。実際の値には署名用のクエリパラメータも含まれますが、ここでは意図的に省略しています。公開された例は仕様に永久に残り、実際の署名は漏洩した認証情報であると同時に、いずれ機能しなくなるリンクでもあるためです。
各画像は 25 MiB、1 回のリクエストの画像は合計 64 MiB に制限されます。
</ParamField>
