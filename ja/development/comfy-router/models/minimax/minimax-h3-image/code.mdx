```python Python
import asyncio
from comfy_sdk import AsyncComfy

# 環境変数 COMFY_API_KEY を読み取ります。
# 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライ時に再利用します。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "minimax/minimax-h3-image",
            {
                "area": "1K",
                "prompt": "A single red maple leaf resting on a plain white background.",
            },
        )
        print("request_id:", handle.request_id)  # モデル ID と合わせれば、別プロセスに必要な情報はこれだけです

        # リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の時間だけ待機します。
        async for update in handle.iter_events():

        # プロバイダー自身のペイロードで、models.run() が返す値と同じです。

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// 環境変数 COMFY_API_KEY を読み取ります。
// 各 submit() 呼び出しは独自の Idempotency-Key を生成し、自動リトライ時に再利用します。
const handle = await comfy.models.submit("minimax/minimax-h3-image", {
  area: "1K",
  prompt: "A single red maple leaf resting on a plain white background.",
});
console.log("requestId:", handle.requestId); // モデル ID と合わせれば、別プロセスに必要な情報はこれだけです

// リクエストが完了するまでポーリングし、サーバーが指定する Retry-After の時間だけ待機します。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// models.run() が返す結果と同じです。失敗またはキャンセル済みのリクエストは、ここで reject されます。
const result = await handle.get();

console.log(result.data);
```

  公開された http(s) URL、または `data:image/<format>;base64,` URI。JPEG、PNG、WEBP、HEIC、HEIF に対応し、各辺は 128 から 8192 px、長辺は短辺の 4 倍以下である必要があります。

</ParamField>

<ParamField body="output_compression" type="integer">

  範囲: `0` から `100`
</ParamField>

<ParamField body="output_format" type="string">
  出力画像の形式。`png`、`jpeg`、`jpg`（デフォルト）、`webp` のいずれか。
</ParamField>

<ParamField body="prompt" type="string" required>
  画像のテキストによる説明。`images` を指定した場合は編集内容の説明となり、参照画像は配列の順に画像 1、画像 2 として扱われます。
</ParamField>

<ParamField body="size" type="string">
  出力サイズ。`auto`（デフォルト）、または `2720x1536` のような `WIDTHxHEIGHT`。固定サイズは面積が 655,360 から 20,000,000 ピクセル、アスペクト比が 1:3 から 3:1 の間である必要があり、`aspect_ratio` や `area` と組み合わせることはできません。
</ParamField>
