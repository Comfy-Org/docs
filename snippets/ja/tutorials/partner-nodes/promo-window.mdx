export const PromoWindow = ({ until = "2026-10-01T00:00:00", children }) => {
  const [show, setShow] = useState(false);
  useEffect(() => {
    setShow(new Date().getTime() < new Date(until).getTime());
  }, [until]);
  if (!show) return null;
  return (
    <div className="my-4 rounded-lg border border-[#49378b]/30 bg-[#49378b]/5 px-4 py-3 text-sm">
      {children}
    </div>
  );
};
