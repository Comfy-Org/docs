import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungen erneut.
async def main():
    async with AsyncComfy() as client:

                "generate_type": "Geometry",

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei die vom Server im Retry-After angegebene Wartezeit abwarten.

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.

asyncio.run(main())

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungen erneut.
const handle = await comfy.models.submit("tencent/hy-3d-3.6", {
  generate_type: "Geometry",
  image_url: "https://example.invalid/tencent/hy-3d-3.6/input.png",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei die vom Server im Retry-After angegebene Wartezeit abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine Anfrage, die fehlgeschlagen oder abgebrochen wurde, wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```
