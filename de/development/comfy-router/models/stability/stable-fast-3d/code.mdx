import { comfy } from "@comfyorg/sdk";

// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungen erneut.

}

  Reichweite: `0.1` bis `1`

</ParamField>

  Mögliche Werte: `none`, `triangle`, `quad`
</ParamField>

<ParamField body="texture_resolution" type="integer">
  Texturgröße. Standardwert 1024.

  Mögliche Werte: `512`, `1024`, `2048`
</ParamField>

<ParamField body="vertex_count" type="integer">
  Zielanzahl der Vertices. -1 behält die Anzahl des Graphen selbst bei. Standardwert -1.

  Reichweite: `-1` bis `20000`
</ParamField>
