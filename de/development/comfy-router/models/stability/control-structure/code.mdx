import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.

        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("stability/control-structure", {
  control_strength: 0.7,
  image: "https://storage.googleapis.com/example-bucket/sketch.png",
  prompt: "a castle on a cliff",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // mit der Modell-ID alles, was ein anderer Prozess braucht

// Pollen, bis die Anfrage abgeschlossen ist, und dabei die vom Server genannte Retry-After-Zeit abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);

## Schema

### Eingaben

<ParamField body="control_strength" type="number">
  Wie genau das Ergebnis dem Quellbild folgt. Standardwert 0.7.

  Bereich: `0` bis `1`
</ParamField>

<ParamField body="image" type="string (uri)" required>
  Eine von Comfy signierte Asset-URL des Quellbildes.

  Format: `uri`
</ParamField>

<ParamField body="negative_prompt" type="string">
  Was aus dem Bild herausgehalten werden soll.
</ParamField>

<ParamField body="output_format" type="string">
  Wird der Wert weggelassen, gilt `png`. `jpeg` und `webp` werden nur ausgeliefert, solange der Ausgabeformat-Schalter für stability in der Bereitstellung aktiv ist; andernfalls werden sie mit einem 400 abgelehnt, bevor eine Aufgabe startet. Das Format ändert nie den Preis.

  Mögliche Werte: `png`, `jpeg`, `webp`
</ParamField>

<ParamField body="prompt" type="string" required>
  Was aus dem Quellbild generiert werden soll.
</ParamField>

<ParamField body="seed" type="integer">
  0 oder ein weggelassener Wert wählt einen zufälligen Seed.

  Bereich: `0` bis `4294967294`
</ParamField>

<ParamField body="style_preset" type="string">
  Wird akzeptiert, aber nicht angewendet.
</ParamField>

### Ausgaben

<ResponseField name="finish_reason" type="string" required>

<ResponseField name="image" type="string (uri)" required>

</ResponseField>

<ResponseField name="seed" type="integer" required>
  Der Seed, mit dem das Bild generiert wurde.
</ResponseField>
