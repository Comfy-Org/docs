```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungen erneut.
const handle = await comfy.models.submit("stability/generate-ultra", {
  aspect_ratio: "16:9",
  prompt: "a lighthouse on a cliff at sunset, oil painting",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier zurückgewiesen.
const result = await handle.get();

console.log(result.data);
```

## Schema

### Eingaben

<ParamField body="aspect_ratio" type="string">
  Nur Text zu Bild. Seitenverhältnis der Ausgabe. Standard: 1:1.

  Mögliche Werte: `1:1`, `16:9`, `9:16`, `21:9`, `9:21`, `3:2`, `2:3`, `5:4`, `4:5`
</ParamField>

<ParamField body="image" type="string (uri)">
  Nur `stability/generate-ultra`. Eine von Comfy signierte Asset-URL eines Quellbilds; wenn Sie sie senden, wechselt der Aufruf zu Bild zu Bild und erfordert `strength`. Bei `stability/generate-core` wird sie mit einem 400 abgelehnt.

  Format: `uri`
</ParamField>

<ParamField body="negative_prompt" type="string">
  Was im Bild nicht vorkommen soll.
</ParamField>

<ParamField body="output_format" type="string">
  Ohne Angabe gilt `png`. `jpeg` und `webp` werden nur bereitgestellt, solange der Schalter für das Ausgabeformat der Bereitstellung aktiv ist; andernfalls werden sie mit einem 400 abgelehnt, bevor eine Aufgabe startet. Das Format ändert nie den Preis.

  Mögliche Werte: `png`, `jpeg`, `webp`
</ParamField>

<ParamField body="prompt" type="string" required>
  Was generiert werden soll.
</ParamField>

<ParamField body="seed" type="integer">
  `0` oder keine Angabe wählt einen zufälligen Seed.

  Bereich: `0` bis `4294967294`
</ParamField>

<ParamField body="strength" type="number">
  Nur für Bild zu Bild mit `stability/generate-ultra`. Wie weit vom Quellbild entfernt wird; `1` ignoriert es. Bei `stability/generate-core` wird der Wert mit einem 400 abgelehnt.

  Bereich: `0` bis `1`
</ParamField>

<ParamField body="style_preset" type="string">
  Stil-Voreinstellung von Stability, zum Beispiel `photographic`.
</ParamField>
