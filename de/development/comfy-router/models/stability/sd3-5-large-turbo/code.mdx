```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.

        # Die Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("stability/sd3.5-large-turbo", {
  aspect_ratio: "16:9",
  prompt: "a lighthouse on a cliff at sunset, oil painting",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // zusammen mit der Model ID alles, was ein anderer Prozess braucht

// Abfragen, bis die Anfrage abgeschlossen ist, dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier zurückgewiesen.
const result = await handle.get();

console.log(result.data);
```

## Schema

### Eingaben

<ParamField body="aspect_ratio" type="string">
  Nur Text zu Bild. Seitenverhältnis der Ausgabe bei etwa einem Megapixel. Standard: 1:1.

  Mögliche Werte: `1:1`, `16:9`, `9:16`, `21:9`, `9:21`, `3:2`, `2:3`, `5:4`, `4:5`
</ParamField>

<ParamField body="cfg_scale" type="number">
  Wie streng das Bild dem Prompt folgt.

  Bereich: `1` bis `10`
</ParamField>

<ParamField body="image" type="string (uri)">
  Nur Bild zu Bild. Eine von Comfy signierte Asset-URL des Quellbildes.

  Format: `uri`
</ParamField>

<ParamField body="mode" type="string">
  Standard ist Text zu Bild. Bild zu Bild erfordert `image` und `strength`.

  Mögliche Werte: `text-to-image`, `image-to-image`
</ParamField>

<ParamField body="negative_prompt" type="string">
  Was im Bild nicht vorkommen soll.
</ParamField>

<ParamField body="output_format" type="string">
  Ohne Angabe wird `png` verwendet. `jpeg` und `webp` werden nur ausgeliefert, solange der Ausgabeformat-Schalter für Stability in der Bereitstellung eingeschaltet ist; andernfalls werden sie mit einem 400 abgelehnt, bevor eine Aufgabe läuft. Das Format ändert nie den Preis.

  Mögliche Werte: `png`, `jpeg`, `webp`
</ParamField>

<ParamField body="prompt" type="string" required>
  Was generiert werden soll.
</ParamField>

<ParamField body="seed" type="integer">
  0 oder ohne Angabe wählt einen zufälligen Seed.

  Bereich: `0` bis `4294967294`
</ParamField>

<ParamField body="strength" type="number">
  Nur Bild zu Bild. Wie weit vom Quellbild abgewichen wird; bei 1 wird es ignoriert.

  Bereich: `0` bis `1`
</ParamField>

<ParamField body="style_preset" type="string">
  Stability-Stilvorlage, zum Beispiel `photographic`.
</ParamField>
