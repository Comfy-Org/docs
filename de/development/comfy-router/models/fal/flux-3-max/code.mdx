import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "fal/flux-3-max",
            {
                "duration": 5,
                "prompt": "A red panda walks along a mossy log in a sunlit forest. Ambient birdsong.",
                "resolution": "720p",
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Model ID alles, was ein anderer Prozess braucht

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("fal/flux-3-max", {
  duration: 5,
  prompt: "A red panda walks along a mossy log in a sunlit forest. Ambient birdsong.",
  resolution: "720p",
});
console.log("requestId:", handle.requestId); // zusammen mit der Model ID alles, was ein anderer Prozess braucht

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier zurückgewiesen.
const result = await handle.get();

console.log(result.data);
```

  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange es noch läuft.

## Schema

### Eingaben

<ParamField body="aspect_ratio" type="string" default="&quot;auto&quot;">
  Seitenverhältnis des generierten Videos. auto entspricht 16:9 bei einem Text-Prompt und folgt andernfalls dem ersten Bild.

  Mögliche Werte: `auto`, `21:9`, `2:1`, `16:9`, `4:3`, `1:1`, `3:4`, `9:16`
</ParamField>

<ParamField body="duration" type="integer" default="5">
  Dauer des generierten Videos in Sekunden, 5 bis 20. fal rechnet jeden Sekundenwert dieses Parameters ab.

  Bereich: `5` bis `20`
</ParamField>

<ParamField body="generate_audio" type="boolean" default="true">
  Synchronisierten Ton generieren. Der Preis bleibt gleich.
</ParamField>

<ParamField body="prompt" type="string" required>
  Text-Prompt, der das Video und seinen Ton beschreibt.
</ParamField>

<ParamField body="reasoning_level" type="string" default="&quot;high&quot;">
  Wie stark das Modell den Prompt vor der Generierung durchdenkt und erweitert. high ist langsamer. Der Preis bleibt gleich.

  Mögliche Werte: `low`, `high`
</ParamField>

<ParamField body="resolution" type="string" default="&quot;720p&quot;">
  Auflösung des generierten Videos. 1080p kostet pro Sekunde mehr als 720p.

  Mögliche Werte: `720p`, `1080p`
</ParamField>

<ParamField body="seed" type="integer">
  Zufälliger Seed. Wird er weggelassen, wird ein zufälliger Seed ausgewählt.
</ParamField>
