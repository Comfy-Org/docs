import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungen erneut.
const handle = await comfy.models.submit("pruna/p-video-2-pro-avatar", {
  input: {
    audio: "https://example.com/speech.mp3",
    image: "https://example.com/portrait.png",
    resolution: "768p",
  },
});
console.log("requestId:", handle.requestId); // zusammen mit der Model ID alles, was ein anderer Prozess braucht

// Fragen Sie ab, bis die Anfrage abgeschlossen ist, und warten Sie dabei das vom Server genannte Retry-After ab.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier zurückgewiesen.
const result = await handle.get();

console.log(result.data);

### Eingaben

<ParamField body="input" type="object" required>
  Parameter für P-Video-2-Pro Avatar. Die Aufgabe wird pro Sekunde des zurückgegebenen Videos abgerechnet, gemessen an der zurückgegebenen Datei und auf eine ganze Sekunde abgerundet, zum Tarif für `resolution`. Der Sicherheitsfilter ist immer aktiv; ein Wert für `enable_safety_filter` in der Anfrage wird ignoriert.
</ParamField>

<ParamField body="input.audio" type="string">

</ParamField>

<ParamField body="input.image" type="string" required>
  Porträt-URI, wird als erstes Frame verwendet. jpg, png oder webp, höchstens 20 MiB. Sie legt das Seitenverhältnis fest.
</ParamField>

<ParamField body="input.prompt" type="string" default="&quot;The person is talking.&quot;">
