```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "meshy/remesh",
            {
                "input_task_id": "0193abcd-0000-0000-0000-000000000000",
                "origin_at": "bottom",
                "resize_height": 1,
                "target_formats": ["glb", "fbx"],
                "target_polycount": 50000,
                "topology": "quad",
            },
        )
        print("request_id:", handle.request_id)  # mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Pollen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage wirft hier den typisierten Router-Fehler.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("meshy/remesh", {
  input_task_id: "0193abcd-0000-0000-0000-000000000000",
  origin_at: "bottom",
  resize_height: 1,
  target_formats: ["glb", "fbx"],
  target_polycount: 50000,
  topology: "quad",
});
console.log("requestId:", handle.requestId); // mit der Modell-ID alles, was ein anderer Prozess benötigt

// Pollen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

```bash cURL
# 2. Pollen, bis der Status COMPLETED ist, und dabei die Retry-After-Sekunden abwarten, die jede Antwort nennt.
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/meshy/remesh/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange es noch läuft.
curl https://api.comfy.org/v2/models/meshy/remesh/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"
```
