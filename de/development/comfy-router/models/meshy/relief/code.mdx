```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("meshy/relief", {
  image_url: "https://example.invalid/meshy/relief/logo.png",
  variant: "low",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Pollen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

## Schema

### Eingaben

<ParamField body="base_thickness_mm" type="number">
  Stärke der Grundplatte in Millimetern, 0,1 bis 10. Bei remove_base true lassen Sie den Wert weg oder senden 0.

  Reichweite: `0` bis `10`
</ParamField>

<ParamField body="bend_angle" type="number" default="60">
  Biegewinkel in Grad. Unter 1 bleibt das Relief flach.

  Reichweite: `0` bis `360`
</ParamField>

<ParamField body="bend_axis" type="string">
  `xz` (Standard, biegt die linken und rechten Kanten) oder `yz` (biegt die oberen und unteren Kanten). Gilt nur, wenn bend_mode convex oder concave ist.
</ParamField>

<ParamField body="bend_mode" type="string">
  `off` (Standard), `convex` oder `concave`.
</ParamField>

<ParamField body="carving_depth_mm" type="number">
  Frästiefe in Millimetern. Der Standardwert ist 6 für variant high und 3 für variant low.

  Reichweite: `0` bis `60`
</ParamField>

<ParamField body="carving_mode" type="string">
  `emboss` (Standard) oder `engrave`.
</ParamField>

<ParamField body="color_mapping" type="boolean" default="false">
  Färbt das Modell anhand des Bildes ein. GLB, OBJ, FBX und USDZ übernehmen die Farbe, STL und 3MF bleiben weiß. Die Kosten ändern sich nicht.
</ParamField>

<ParamField body="image_url" type="string" required>
  Eine öffentliche URL oder ein Data-URI eines Bildes im Format jpg, jpeg, png oder webp. Höchstens 20 MB und mindestens 32 x 32 Pixel. Transparente Bereiche werden mit Schwarz gefüllt.
</ParamField>

<ParamField body="mesh_resolution" type="integer" default="1024">
  Anzahl der Punkte entlang der längeren Seite. Größere Werte ergeben deutlich größere Dateien (etwa 150 MB GLB und 300 MB STL bei 2048).

  Reichweite: `512` bis `2048`
</ParamField>

<ParamField body="removal_threshold" type="number" default="0.01">
  Schwellenwert für die Hintergrundentfernung. Gilt nur, wenn remove_base true ist.

  Reichweite: `0.01` bis `1`
</ParamField>

<ParamField body="remove_base" type="boolean" default="false">
  Schneidet den Hintergrund weg, sodass das Relief keine Grundplatte hat.
</ParamField>

<ParamField body="should_generate_model" type="boolean" default="true">
  Auf false setzen, um nur eine Tiefenkarte zu erhalten, ohne Modell und ohne Vorschau. Die Kosten ändern sich nicht.
</ParamField>

<ParamField body="size_mm" type="number" default="60">
  Länge der längeren Seite in Millimetern.

  Reichweite: `20` bis `250`
</ParamField>

<ParamField body="target_formats" type="string[]">
  Die zu generierenden Modellformate. Eines oder mehrere aus glb, obj, fbx, stl, usdz und 3mf. Der Standard ist glb und stl.
</ParamField>

<ParamField body="variant" type="string">
  `low` (Standard) liefert eine geringe Tiefe und feinere Details, für Laser- oder CNC-Arbeiten. `high` liefert ein tieferes, skulpturales Relief für den 3D-Druck. Es legt außerdem den Standardwert für carving_depth_mm fest.
</ParamField>
