import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "meshy/rigging",
            {
                "height_meters": 1.8,
                "input_task_id": "0193abcd-0000-0000-0000-000000000000",
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("meshy/rigging", {
  height_meters: 1.8,
  input_task_id: "0193abcd-0000-0000-0000-000000000000",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

# 1. Absenden. Router antwortet mit 201 und request_id, status_url, response_url und cancel_url.
curl https://api.comfy.org/v2/models/meshy/rigging/requests \
  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"height_meters\": 1.8, \"input_task_id\": \"0193abcd-0000-0000-0000-000000000000\"}"

# 2. Abfragen, bis der Status COMPLETED ist, und dabei die in jeder Antwort genannten Retry-After-Sekunden abwarten.
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/meshy/rigging/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange es noch läuft.
curl https://api.comfy.org/v2/models/meshy/rigging/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"

## Schema

### Eingaben

<ParamField body="height_meters" type="number" default="1.7">
  Die ungefähre Höhe des Charaktermodells in Metern. Muss eine positive Zahl sein.

  Bereich: `0` bis `…`
</ParamField>

<ParamField body="include_fingers" type="boolean" default="false">
  Ob den Händen des Charakters Fingerknochen hinzugefügt werden. Nur Biped-Rig; mit texture_image_url, einem Quadruped-Rig oder einer input_task_id mit Quad-Topologie lehnt Meshy die Anfrage mit 400 ab, und ein Modell, das nur ohne Finger geriggt werden kann, mit 422 (keine Abrechnung). Fingerknochen ändern nichts an den Credit-Kosten. Standardwert ist false.
</ParamField>

<ParamField body="input_task_id" type="string">
  Die Eingabe-Aufgabe, die geriggt werden soll. Erforderlich, wenn model_url nicht angegeben ist.
</ParamField>

<ParamField body="model_url" type="string">
  Eine öffentlich erreichbare URL oder Data-URI zu einer texturierten humanoiden GLB-Datei. Erforderlich, wenn input_task_id nicht angegeben ist.
</ParamField>

<ParamField body="skeleton_template" type="string" default="&quot;meshyrig&quot;">
  Das Skelett für den geriggten Charakter, das dessen Knochennamen und Hierarchie festlegt. Nur Biped-Rig. Bei einem anderen Wert als meshyrig (oder include_fingers true) liegen die Ausgabedateien in Metern vor und verwenden die Knochen dieses Skeletts statt der standardmäßigen 24-Knochen-Dateien von meshyrig in Zentimetern; eine Animationsaufgabe, die aus diesem Rig erstellt wird, verwendet dasselbe Skelett. Jeder andere Wert wird abgelehnt. Standardwert ist meshyrig.

  Mögliche Werte: `meshyrig`, `mixamo`, `ue5_mannequin`, `max_biped`, `unity_humanoid`, `maya_human_ik`, `blender_rigify_basic_human`, `cc5_hd_neutral_base`, `daz_genesis9`
</ParamField>

<ParamField body="texture_image_url" type="string">
  Die UV-entfaltete Grundfarben-Textur des Modells. Öffentlich erreichbare URL oder Data-URI. Unterstützt das Format .png.
</ParamField>
