import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "meshy/animations",
            {
                "action_id": 92,
                "post_process": {
                    "fps": 60,
                    "operation_type": "change_fps",
                },
                "rig_task_id": "0193abcd-0000-0000-0000-000000000000",
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Model-ID alles, was ein anderer Prozess braucht

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("meshy/animations", {
  action_id: 92,
  post_process: {
    fps: 60,
    operation_type: "change_fps",
  },
  rig_task_id: "0193abcd-0000-0000-0000-000000000000",
});
console.log("requestId:", handle.requestId); // zusammen mit der Model-ID alles, was ein anderer Prozess braucht

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier zurückgewiesen.
const result = await handle.get();

console.log(result.data);
```

# 1. Absenden. Router antwortet mit 201 und request_id, status_url, response_url und cancel_url.
curl https://api.comfy.org/v2/models/meshy/animations/requests \
  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"action_id\": 92, \"post_process\": {\"fps\":60,\"operation_type\":\"change_fps\"}, \"rig_task_id\": \"0193abcd-0000-0000-0000-000000000000\"}"

# 2. Abfragen, bis der Status COMPLETED ist, und dabei die in jeder Antwort genannten Retry-After-Sekunden abwarten.
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/meshy/animations/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange er noch läuft.
curl https://api.comfy.org/v2/models/meshy/animations/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"

## Schema

### Eingaben

<ParamField body="action_id" type="integer" required>
  Die Kennung der anzuwendenden Animationsaktion.

<ParamField body="post_process" type="object">

</ParamField>

<ParamField body="post_process.fps" type="integer" default="30">

</ParamField>

<ParamField body="post_process.operation_type" type="string" required>
  Der Typ der auszuführenden Operation.

  Mögliche Werte: `change_fps`, `fbx2usdz`, `extract_armature`
</ParamField>

<ParamField body="rig_task_id" type="string" required>
  Die ID einer erfolgreich abgeschlossenen Rigging-Aufgabe (aus POST /openapi/v1/rigging). Die Figur aus dieser Aufgabe wird animiert.
</ParamField>
