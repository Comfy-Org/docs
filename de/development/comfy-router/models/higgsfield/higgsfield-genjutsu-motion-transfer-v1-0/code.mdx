import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungsversuchen erneut.
const handle = await comfy.models.submit("higgsfield/higgsfield-genjutsu-motion-transfer-v1.0", {
  image_urls: ["https://cdn.example.com/character.jpg"],
  prompt: "Keep the camera move and recast the dancer as the character in the reference image",

  video_url: "https://cdn.example.com/source.mp4",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist; dabei die vom Server genannten Retry-After-Sekunden abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage führt hier zu einer Ablehnung.
