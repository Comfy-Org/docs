import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("pika/pika-sfx", {
  duration: 2,
  prompt: "a wooden door creaking open",
});
console.log("requestId:", handle.requestId); // with the model ID, all another process needs

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei die vom Server angegebene Retry-After-Zeit abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine Anfrage, die fehlgeschlagen oder abgebrochen wurde, wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
