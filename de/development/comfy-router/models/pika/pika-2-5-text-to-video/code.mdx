```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "pika/pika-2.5-text-to-video",
            {
                "duration_s": 5,
                "prompt": "A paper boat drifting down a rain-soaked city street at dusk",
                "resolution": "720p",
            },
        )
        print("request_id:", handle.request_id)  # mit der Model ID alles, was ein anderer Prozess braucht

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt seinen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("pika/pika-2.5-text-to-video", {
  duration_s: 5,
  prompt: "A paper boat drifting down a rain-soaked city street at dusk",
  resolution: "720p",
});
console.log("requestId:", handle.requestId); // mit der Model ID alles, was ein anderer Prozess braucht

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

```bash cURL
# 1. Absenden. Router antwortet mit 201 und request_id, status_url, response_url und cancel_url.
curl https://api.comfy.org/v2/models/pika/pika-2.5-text-to-video/requests \
  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"duration_s\": 5, \"prompt\": \"A paper boat drifting down a rain-soaked city street at dusk\", \"resolution\": \"720p\"}"

# 2. Abfragen, bis status COMPLETED ist, und dabei die in jeder Antwort genannten Retry-After-Sekunden abwarten.
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/pika/pika-2.5-text-to-video/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange es noch läuft.
curl https://api.comfy.org/v2/models/pika/pika-2.5-text-to-video/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"
```
