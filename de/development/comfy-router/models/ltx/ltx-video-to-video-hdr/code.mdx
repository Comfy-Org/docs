```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "ltx/ltx-video-to-video-hdr",
            {
                "output_colorspace": "acescg",
                "video_uri": "https://test-videos.co.uk/vids/sintel/mp4/h264/720/Sintel_720_10s_1MB.mp4",
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Abfragen, bis die Anfrage abgeschlossen ist, wobei das vom Server angegebene Retry-After abgewartet wird.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("ltx/ltx-video-to-video-hdr", {
  output_colorspace: "acescg",
  video_uri: "https://test-videos.co.uk/vids/sintel/mp4/h264/720/Sintel_720_10s_1MB.mp4",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, wobei das vom Server angegebene Retry-After abgewartet wird.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

## Schema

### Eingaben

<ParamField body="output_colorspace" type="string" default="&quot;acescg&quot;">
  Farbraum der OpenEXR-Bildsequenz unter `result.exr_frames_url`. Er wirkt sich nicht auf `result.video_url` aus.

  Mögliche Werte: `acescg`, `acescct`, `srgb_linear`, `aces2065-1`
</ParamField>

<ParamField body="video_uri" type="string" required>
  Eingabevideo als öffentlich zugängliche HTTPS-URL oder als Base64-Daten-URI. H.264 oder HEVC in MP4, MOV oder MKV, bis zu 200 MB per URL. Breite und Höhe müssen in beiden Ausrichtungen innerhalb von 3840x2160 liegen.
</ParamField>
