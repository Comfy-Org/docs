import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungen erneut.
const handle = await comfy.models.submit("typesafe/jev-1.14.0", {
  questions: {
    department: {
      criteria: {
        billing: "Payments, invoicing, refunds",
        technical: "Bugs, outages, integrations",
      },
      instructions: "Which team should handle this?",
      type: "choice",
    },
    is_urgent: {
      instructions: "Does this convey urgency?",
      type: "noul",
    },
  },
  state: "Help! My payouts have been failing for 3 days.",
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei die vom Server genannte Retry-After-Zeit abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.

## Schema

### Eingaben

<ParamField body="assets" type="object">
  Die Bilder in der Anfrage, adressiert über eine Referenzzeichenfolge. Jede Referenz muss an irgendeiner Stelle in `state`, `instructions` oder `criteria` vorkommen. Jedes Bild ist ein Standbild im PNG- oder JPEG-Format mit 280 bis 1568 Pixeln pro Seite und höchstens 16 MiB nach dem Dekodieren; TypeSafe lehnt einen Aufruf mit mehr als 32 MiB insgesamt ab.
</ParamField>

<ParamField body="model" type="string">
  Die angeheftete Jev-Version. Optional: Router füllt sie aus der Modell-ID im Pfad. Die Aliase `jev-latest` und `jev-preview` werden nicht akzeptiert.
</ParamField>

<ParamField body="questions" type="object" required>
  Die Fragen, die beantwortet werden sollen, adressiert über eine von Ihnen gewählte ID. Die Antwort auf jede Frage wird unter derselben ID zurückgegeben.
</ParamField>

<ParamField body="state" type="object" required>
  Der Inhalt, der ausgewertet werden soll: eine Zeichenfolge oder strukturierte Daten (ein Objekt oder Array) mit bis zu 64 Verschachtelungsebenen. Eine Asset-Referenz an beliebiger Stelle darin wird durch das entsprechende Bild ersetzt.
</ParamField>
