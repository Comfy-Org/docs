import { comfy } from "@comfyorg/sdk";

});

// Abfragen, bis die Anfrage abgeschlossen ist; dabei wird die vom Server genannte Retry-After-Zeit abgewartet.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
