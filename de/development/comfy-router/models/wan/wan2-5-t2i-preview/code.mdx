```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "wan/wan2.5-t2i-preview",
            {
                "input": {
                    "prompt": "A single red maple leaf on a plain white background.",
                },
                "parameters": {
                    "n": 1,
                    "size": "1280*1280",
                },
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Abfragen, bis die Anfrage abgeschlossen ist; dabei die vom Server genannten Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die Payload des Anbieters selbst, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Liest COMFY_API_KEY aus der Umgebung.
// Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn für automatische Wiederholungsversuche erneut.
const handle = await comfy.models.submit("wan/wan2.5-t2i-preview", {
  input: {
    prompt: "A single red maple leaf on a plain white background.",
  },
  parameters: {
    n: 1,
    size: "1280*1280",
  },
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist; dabei die vom Server genannten Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.
const result = await handle.get();

console.log(result.data);
```

```bash cURL
# 1. Absenden. Router antwortet mit 201 und request_id, status_url, response_url und cancel_url.
curl https://api.comfy.org/v2/models/wan/wan2.5-t2i-preview/requests \
  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"input\": {\"prompt\":\"A single red maple leaf on a plain white background.\"}, \"parameters\": {\"n\":1,\"size\":\"1280*1280\"}}"

# 2. Abfragen, bis der Status COMPLETED ist; dabei jeweils die in der Antwort genannten Retry-After-Sekunden abwarten.
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/wan/wan2.5-t2i-preview/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. Abholen. 200 mit der nativen Ausgabe des Modells, 202 mit dem Status-Body, solange die Anfrage noch läuft.
curl https://api.comfy.org/v2/models/wan/wan2.5-t2i-preview/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"
```

### Eingaben

<ParamField body="input" type="object" required>
  Geben Sie grundlegende Informationen ein, etwa Prompt-Wörter usw.

<ParamField body="input.negative_prompt" type="string">
  Negative Prompt-Wörter, die Inhalte beschreiben, die Sie im Bild nicht sehen möchten. Ab 500 Zeichen kürzt das Modell den Text, statt ihn abzulehnen. Er zählt auf dasselbe Eingabe-Budget wie prompt.
</ParamField>

<ParamField body="input.prompt" type="string" required>
  Positive Prompt-Wörter, die erwartete Bildelemente und visuelle Merkmale beschreiben. Unterstützt Chinesisch und Englisch. Ab 2.000 Zeichen kürzt das Modell den Prompt, statt ihn abzulehnen. Der Anbieter lehnt eine Anfrage ab, deren prompt und negative_prompt zusammen länger als etwa 60.850 Bytes UTF-8-Text sind (etwa 20.000 chinesische Zeichen), und antwortet mit InvalidParameter `Range of input length should be [1, 61440]`.
</ParamField>

<ParamField body="model" type="string">
  Die ID des Modells, das für die Text-zu-Bild-Generierung aufgerufen wird. Auf dieser Komponente NICHT eingeschränkt: Comfy Router füllt sie aus dem Pfadsegment `{model}` von `POST /v2/models/wan/{model}`. Ein direkter v1-Aufruf an `POST /proxy/wan/api/v1/services/aigc/text2image/image-synthesis` MUSS sie angeben, und das Enum der akzeptierten Schreibweisen liegt auf der eigenen Komponente dieser Operation, `WanImageGenerationRequest`.
</ParamField>

<ParamField body="parameters" type="object">
  Parameter der Bildverarbeitung
</ParamField>

<ParamField body="parameters.n" type="integer" default="4">
  Anzahl der generierten Bilder. Bereich 1-4, Standard ist 4

<ParamField body="parameters.prompt_extend" type="boolean" default="true">
  Aktiviert die intelligente Umschreibung des Prompts. Standard ist true
</ParamField>

<ParamField body="parameters.seed" type="integer">
  Seed für die Zufallszahlen zur Steuerung der Zufälligkeit. Bereich [0, 2147483647]

  Bereich: `0` bis `2147483647`

<ParamField body="parameters.size" type="string" default="&quot;1280*1280&quot;">
  Die Auflösung des Ausgabebilds im Format Breite*Höhe. Standard ist 1280*1280. Die API akzeptiert eine Pixelanzahl zwischen 1638400 (1280*1280) und 2073600 (1440*1440) bei einem Seitenverhältnis zwischen 1:4 und 4:1, daher ist 768*2700 gültig
