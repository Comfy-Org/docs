import asyncio
from comfy_sdk import AsyncComfy

# Liest COMFY_API_KEY aus der Umgebung.
# Jeder submit()-Aufruf erzeugt einen eigenen Idempotenzschlüssel und nutzt ihn bei automatischen Wiederholungsversuchen erneut.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "wan/happyhorse-1.0-t2v",
            {
                "input": {
                    "prompt": "a single red maple leaf falling onto still water, slow motion, shallow depth of field",
                },
                "parameters": {
                    "resolution": "720P",
                    "duration": 5,
                },
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei die Retry-After-Sekunden abwarten, die der Server nennt.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die Payload des Anbieters selbst, derselbe Wert, den models.run() zurückgibt.
        # Eine Anfrage, die fehlgeschlagen oder abgebrochen wurde, löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print("video:", result["output"]["video_url"])

asyncio.run(main())
