API-Referenz für Wan 2.5 I2V Preview. Wan 2.5 Bild zu Video verwandelt ein Standbild als erstes Einzelbild in ein kurzes Video, wobei die Bewegung und die Kamerabewegung durch den Prompt beschrieben werden.

# Jeder submit()-Aufruf erzeugt einen eigenen Idempotency-Key und verwendet ihn bei automatischen Wiederholungen erneut.

                "input": {

                "parameters": {
                    "resolution": "720P",
                    "duration": 5,
                },
            },
        )
        print("request_id:", handle.request_id)  # zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

        # Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Die eigene Payload des Anbieters, derselbe Wert, den models.run() zurückgibt.
        # Eine fehlgeschlagene oder abgebrochene Anfrage löst hier den typisierten Router-Fehler aus.
        result = await handle.get()

    print("video:", result["output"]["video_url"])

asyncio.run(main())

const handle = await comfy.models.submit<Result>("wan/wan2.5-i2v-preview", {
  input: {
    prompt: "the robot lowers its raised arm and steps toward the camera, the flat orange backdrop holds steady",
    img_url: "https://raw.githubusercontent.com/Comfy-Org/workflow_templates/main/input/pink_robot_front.png",
  },
  parameters: {
    resolution: "720P",
    duration: 5,
  },
});
console.log("requestId:", handle.requestId); // zusammen mit der Modell-ID alles, was ein anderer Prozess benötigt

// Abfragen, bis die Anfrage abgeschlossen ist, und dabei das vom Server genannte Retry-After abwarten.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Dasselbe Ergebnis, das models.run() zurückgibt. Eine fehlgeschlagene oder abgebrochene Anfrage wird hier abgelehnt.

if (result.kind !== "json") throw new Error("expected a JSON result");

Für wan2.6 T2V/R2V (kein 480P):
