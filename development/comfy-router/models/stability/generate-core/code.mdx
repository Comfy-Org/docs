  Format: `uri`
</ParamField>

<ParamField body="negative_prompt" type="string">
  What to keep out of the image.
</ParamField>

<ParamField body="output_format" type="string">
  Only `png` is returned.

  Possible values: `png`
</ParamField>

<ParamField body="prompt" type="string" required>
  What to generate.
</ParamField>

<ParamField body="seed" type="integer">
  0 or omitted picks a random seed.

  Range: `0` to `4294967294`
</ParamField>

<ParamField body="strength" type="number">
  `stability/generate-ultra` image-to-image only. How far to move from the source image; 1 ignores it.

  Range: `0` to `1`
</ParamField>

<ParamField body="style_preset" type="string">
