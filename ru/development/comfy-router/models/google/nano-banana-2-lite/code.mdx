from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменных окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "vertexai/gemini-3.1-flash-lite-image",
            {
                "contents": [
                    {
                        "role": "user",
                        "parts": [
                            {
                                "text": "a single red maple leaf on a plain white background, studio lighting",
                            },
                        ],
                    },
                ],
                "generationConfig": {
                    "responseModalities": ["IMAGE"],
                    "imageConfig": {
                        "aspectRatio": "1:1",
                    },
                },
            },
        )
        print("request_id:", handle.request_id)  # вместе с ID модели это всё, что нужно другому процессу

        # Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, которое возвращает models.run().
        # Запрос, завершившийся ошибкой или отменённый, вызывает здесь типизированную ошибку Router.

    print("image (base64):", result["candidates"][0]["content"]["parts"][0]["inlineData"]["data"])

## Примеры

### Вход

```json
{
  "contents": [
    {
      "role": "user",
      "parts": [
        {
          "text": "a single red maple leaf on a plain white background, studio lighting"
        }
      ]
    }
  ],
  "generationConfig": {
    "responseModalities": [
      "IMAGE"
    ],
    "imageConfig": {
      "aspectRatio": "1:1"
    }
  }
}
```

### Выход

```json
{
  "candidates": [
    {
      "content": {
        "role": "model",
        "parts": [
          {
            "inlineData": {
              "mimeType": "image/png",
              "data": "PGJhc2U2ND4="
            }
          }
        ]
      },
      "finishReason": "STOP"
    }
  ],
  "usageMetadata": {
    "promptTokenCount": 12,
    "candidatesTokenCount": 1290
  }
}
```

По умолчанию сгенерированные части изображения содержат байты base64 в `inlineData.data` и тип медиа в `inlineData.mimeType`. Декодируйте байты и сохраните их в файл. При `uploadImagesToStorage: true` загруженные изображения вместо этого используют `fileData.fileUri` для подписанного URL и `fileData.mimeType` для типа медиа. Скачайте эти изображения до истечения срока действия URL, то есть в течение 24 часов после их создания. Если загрузка не удалась, изображение остаётся inline, поэтому проверяйте каждую часть на наличие `inlineData` или `fileData`; также могут присутствовать текстовые части, и изображение не обязательно будет первой частью. Путь `candidates[0].content.parts[0].inlineData.data` в примерах быстрого старта выше читает единственную inline-часть примера ответа на этой странице; при работе с реальным ответом вместо индексации по позиции 0 просматривайте `parts` в поисках нужного ключа.
