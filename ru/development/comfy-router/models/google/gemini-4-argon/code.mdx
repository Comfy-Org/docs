import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "vertexai/gemini-4-argon",
            {
                "contents": [
                    {
                        "parts": [
                            {
                                "text": "Describe a robot learning to paint, in two sentences.",

                    },

        )

        # Опрашивать, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.

        },

    },

// Опрашивать, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// То же значение, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.

<ParamField body="generationConfig.topP" type="number" default="0.95">
  Если задано, используется ядерная выборка.
Top-P изменяет то, как модель выбирает токены для вывода. Токены выбираются от наиболее вероятных (см. top-K) до наименее вероятных, пока сумма их вероятностей не станет равной значению top-P. Укажите меньшее значение для менее случайных ответов и большее для более случайных ответов.

  Диапазон: от `0` до `1`
