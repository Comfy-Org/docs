import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его для автоматических повторов.
async def main():

        # Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
        async for update in handle.iter_events():

        # Собственный payload провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, вызывает здесь типизированную ошибку Router.

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.

for await (const update of handle.events()) {

}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.

  Идентификатор модели Gemini: `gemini-omni-1.2-flash` или `gemini-omni-1.1-flash` (общедоступные), либо устаревший `gemini-omni-flash-preview`. На маршруте Comfy Router `POST /v2/models/gemini-interactions/{model}` он берётся из пути и может быть опущен. Именно эти три написания (набор, который Comfy Router адресует как `gemini-interactions/<model>`) обслуживает эта операция (supportedGeminiInteractionModels); здесь они перечислены явно, а не ограничены перечислением, потому что прокси сам проверяет модель и отвечает собственным 400 на написание, которое он не обслуживает.
</ParamField>

<ParamField body="previous_interaction_id" type="string">
