import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменных окружения.
# Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его при автоматических повторных попытках.
async def main():

        # Собственная полезная нагрузка провайдера: то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, здесь вызывает типизированную ошибку Router.

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменных окружения.
// Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его при автоматических повторных попытках.
const handle = await comfy.models.submit("stability/edit-inpaint", {
  image: "https://storage.googleapis.com/example-bucket/source.png",
  mask: "https://storage.googleapis.com/example-bucket/mask.png",
  prompt: "a red door",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.
const result = await handle.get();

console.log(result.data);
