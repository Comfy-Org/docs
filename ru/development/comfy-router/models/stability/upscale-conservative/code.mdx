import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "stability/upscale-conservative",
            {
                "creativity": 0.35,
                "image": "https://storage.googleapis.com/example-bucket/source.png",
                "prompt": "a detailed photo of a cat",
                "seed": 12345,
            },
        )
        print("request_id:", handle.request_id)  # вместе с model ID это всё, что нужно другому процессу

        # Опрашивает до завершения запроса, выдерживая Retry-After, который указывает сервер.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный ответ провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("stability/upscale-conservative", {
  creativity: 0.35,
  image: "https://storage.googleapis.com/example-bucket/source.png",
  prompt: "a detailed photo of a cat",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // вместе с model ID это всё, что нужно другому процессу

// Опрашивает до завершения запроса, выдерживая Retry-After, который указывает сервер.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.
const result = await handle.get();

console.log(result.data);

## Схема

### Входы

<ParamField body="creativity" type="number">
  Сколько деталей добавлять. По умолчанию 0.35.

  Диапазон: от `0.2` до `0.5`
</ParamField>

<ParamField body="image" type="string (uri)" required>
  Подписанный Comfy URL ресурса исходного изображения.

  Формат: `uri`
</ParamField>

<ParamField body="negative_prompt" type="string">
  Что не должно попасть в изображение после upscale.
</ParamField>

<ParamField body="output_format" type="string">
  Если параметр не указан, используется `png`. `jpeg` и `webp` отдаются только пока в развёртывании включён переключатель формата вывода stability; иначе они отклоняются с кодом 400 ещё до запуска любой задачи. Формат никогда не влияет на цену.

  Возможные значения: `png`, `jpeg`, `webp`
</ParamField>

<ParamField body="prompt" type="string" required>
  Что должно быть на изображении после upscale.
</ParamField>

<ParamField body="seed" type="integer">
  `0` или отсутствие значения выбирает случайный seed.

  Диапазон: от `0` до `4294967294`
</ParamField>
