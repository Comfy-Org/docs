import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "stability/edit-erase",
            {
                "grow_mask": 5,
                "image": "https://storage.googleapis.com/example-bucket/source.png",
                "mask": "https://storage.googleapis.com/example-bucket/mask.png",
                "seed": 12345,
            },
        )
        print("request_id:", handle.request_id)  # вместе с ID модели это всё, что нужно другому процессу

        # Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, которое возвращает models.run().
        # Запрос, завершившийся ошибкой или отменённый, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("stability/edit-erase", {
  grow_mask: 5,
  image: "https://storage.googleapis.com/example-bucket/source.png",
  mask: "https://storage.googleapis.com/example-bucket/mask.png",
  seed: 12345,
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, который возвращает models.run(). Запрос, завершившийся ошибкой или отменённый, отклоняется здесь.
const result = await handle.get();

console.log(result.data);

## Схема

### Входы

<ParamField body="grow_mask" type="integer">
  На сколько пикселей расширить маску. По умолчанию 5.

  Диапазон: `0` до `20`
</ParamField>

<ParamField body="image" type="string (uri)" required>
  Подписанный Comfy URL ресурса исходного изображения.

  Формат: `uri`
</ParamField>

<ParamField body="mask" type="string (uri)">
  Подписанный Comfy URL ресурса маски. Белый цвет обозначает область, которую нужно стереть. Если значение не указано, маской служит альфа-канал самого изображения.

  Формат: `uri`
</ParamField>

<ParamField body="output_format" type="string">
  Если значение не указано, используется `png`. `jpeg` и `webp` выдаются только пока в развёртывании включён переключатель форматов вывода stability; в противном случае они отклоняются с кодом 400 ещё до запуска любой задачи. Формат никогда не влияет на цену.

  Возможные значения: `png`, `jpeg`, `webp`
</ParamField>

<ParamField body="seed" type="integer">
  0 или отсутствие значения означает случайный seed.

  Диапазон: `0` до `4294967294`
</ParamField>
