То же тело запроса, отправленное на `POST https://api.comfy.org/v2/models/cartwheel/swing-sketch/requests`. Router отвечает `201` с `request_id` сразу после того, как запуск принят, а результат забирается, когда он готов, из этого процесса или из другого. [Доставка через очередь](/ru/development/comfy-router/queue) описывает статус, отмену и получение результата.

```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его для автоматических повторов.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "cartwheel/swing-sketch",
            {
                "exportSettings": {
                    "characterID": "char-upload-MHR",
                    "exportType": "glb",
                    "forward": "-Z",
                    "frameRate": 30,
                    "frameStepSize": 1,
                    "up": "Y",
                },
                "prompts": ["a person waves with the right hand"],
                "swingSketchProperties": {
                    "duration": 4,
                },
            },
        )
        print("request_id:", handle.request_id)  # вместе с ID модели это всё, что нужно другому процессу

        # Опрашивайте, пока запрос не завершится, выдерживая Retry-After, который указывает сервер.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственная полезная нагрузка провайдера, то же значение, которое возвращает models.run().
        # Запрос, завершившийся ошибкой или отменённый, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его для автоматических повторов.
const handle = await comfy.models.submit("cartwheel/swing-sketch", {
  exportSettings: {
    characterID: "char-upload-MHR",
    exportType: "glb",
    forward: "-Z",
    frameRate: 30,
    frameStepSize: 1,
    up: "Y",
  },
  prompts: ["a person waves with the right hand"],
  swingSketchProperties: {
    duration: 4,
  },
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выдерживая Retry-After, который указывает сервер.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, который возвращает models.run(). Запрос, завершившийся ошибкой или отменённый, отклоняется здесь.
const result = await handle.get();

console.log(result.data);
```
