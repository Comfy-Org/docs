import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его для автоматических повторов.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "kling/kling-4.0",
            {
                "contents": [
                    {
                        "text": "The fox turns toward the camera.",
                        "type": "prompt",
                    },
                    {
                        "type": "first_frame",
                        "url": "https://example.com/fox.png",
                    },
                ],
                "settings": {
                    "duration": 3,
                    "resolution": "720p",
                },
            },
        )
        print("request_id:", handle.request_id)  # вместе с ID модели это всё, что нужно другому процессу

        # Опрашивайте, пока запрос не завершится, выжидая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его для автоматических повторов.
const handle = await comfy.models.submit("kling/kling-4.0", {

      text: "The fox turns toward the camera.",
      type: "prompt",
    },
    {
      type: "first_frame",
      url: "https://example.com/fox.png",
    },
  ],
  settings: {
    duration: 3,
    resolution: "720p",
  },
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выжидая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.
const result = await handle.get();

console.log(result.data);

## Схема

### Входы

<ParamField body="contents" type="object[]">
  Промпт и входные медиаданные. Если набор содержит только элементы «prompt», «first_frame», «middle_frame» и «last_frame», выбирается изображение в видео (промпт и до 10 кадров); если содержит любой из «refer_image», «feature_video», «base_video», «element» или «voice», выбирается omni video (до 15 входов, максимум 10 изображений, 5 видео, 7 элементов и 3 голоса). Запрос с «feature_video», «base_video» или элементом-видеоперсонажем тарифицируется по ставке за видеовход.
</ParamField>

<ParamField body="contents[].element_id" type="string">
  ID элемента, когда type равно «element».
</ParamField>

<ParamField body="contents[].id" type="string">
  Ссылочный ID для этого входа, используется как @id в промпте. Должен быть уникальным в пределах задачи. Обязателен для «element» и «voice».
</ParamField>

<ParamField body="contents[].text" type="string">
  Текстовый промпт, когда type равно «prompt». До 26 000 английских символов или 11 000 неанглийских; comfy-api допускает от 1 до 26 000 символов, а сам Kling отклоняет более длинный неанглийский текст. Ссылайтесь на другие входы как @id.
</ParamField>

<ParamField body="contents[].type" type="string" required>
  Тип входа. Изображение в видео принимает только «prompt», «first_frame», «middle_frame» и «last_frame» и отклоняет любой другой тип. Omni video также принимает «refer_image», «feature_video» (референсное видео), «base_video» (видео для редактирования, не более одного), «element» и «voice».

  Возможные значения: `prompt`, `first_frame`, `middle_frame`, `last_frame`, `refer_image`, `feature_video`, `base_video`, `element`, `voice`
</ParamField>

<ParamField body="contents[].url" type="string">
  URL изображения или видео, когда type равно кадру, «refer_image», «feature_video» или «base_video». Изображения также могут быть в Base64. Изображения: .jpg, .jpeg или .png до 30MB, не менее 300px по каждой стороне, с соотношением сторон от 2:5 до 5:2. Видео: .mp4 или .mov до 300MB длительностью от 3 до 30 секунд.
</ParamField>

<ParamField body="contents[].voice_id" type="string">
  ID голоса, когда type равно «voice». Пресетный голос Kling или пользовательский голос.
</ParamField>

<ParamField body="options" type="object">
  Общие настройки, например параметры водяного знака. `options.callback_url` и `options.external_task_id` не поддерживаются на Kling 4.0; Comfy Router отклоняет запрос, задающий любой из них внутри `options`. Верхнеуровневые `callback_url` или `external_task_id` не входят в эту схему и удаляются до того, как запрос достигнет Kling.
</ParamField>

<ParamField body="options.watermark_info" type="object">
  Нужно ли одновременно генерировать результаты с водяным знаком. Пользовательские водяные знаки не поддерживаются. Явный null считается отсутствующим, как это делают проверка Kling 4.0 в /proxy и сам Kling.
</ParamField>

<ParamField body="options.watermark_info.enabled" type="boolean">
  true означает генерировать результат с водяным знаком, false означает не генерировать. По умолчанию false.
</ParamField>

<ParamField body="prompt" type="string">
  Промпт для текст в видео. Обязателен, когда `contents` отсутствует или пуст, и отклоняется вместе с непустым `contents`. До 26 000 английских символов или 11 000 неанглийских; comfy-api допускает от 1 до 26 000 символов, а сам Kling отклоняет более длинный неанглийский текст.
</ParamField>

<ParamField body="settings" type="object" required>
  Настройки выходных данных, такие как разрешение, соотношение сторон и длительность.
</ParamField>

<ParamField body="settings.aspect_ratio" type="string">
  Соотношение сторон (ширина:высота) сгенерированных кадров. Одно из «21:9», «16:9», «9:16», «1:1», «4:3» или «3:4» для текст в видео (по умолчанию «16:9»), они же или «adaptive» для omni video (по умолчанию «adaptive»). Изображение в видео следует за изображениями. Явный null считается отсутствующим, как это делают проверка Kling 4.0 в /proxy и сам Kling.

  Возможные значения: `21:9`, `16:9`, `9:16`, `1:1`, `4:3`, `3:4`, `adaptive`
</ParamField>

<ParamField body="settings.duration" type="integer">
  Длительность видео в секундах, от 3 до 30. Обязательна, если `contents` не содержит «base_video»: при редактировании её можно опустить, чтобы сохранить длительность базового видео. Такой вызов приостанавливается и оценивается по максимуму в 30 секунд при запрошенном разрешении, это верхняя граница, а не цена, и списание идёт по цене, которую Kling сообщает для завершённой задачи.

  Диапазон: от `3` до `30`
</ParamField>

<ParamField body="settings.dynamic_range" type="string">
  Динамический диапазон сгенерированного видео. Одно из «sdr» или «hdr». По умолчанию «sdr». «hdr» требует разрешения «1080p» или «4k» и отклоняется при «720p». HDR-видео стоит дороже за каждую выходную секунду. Явный null считается отсутствующим, как это делают проверка Kling 4.0 в /proxy и сам Kling.

  Возможные значения: `sdr`, `hdr`
</ParamField>

<ParamField body="settings.resolution" type="string" required>
  Разрешение сгенерированного видео. Одно из «720p», «1080p» или «4k». Обязательно.

  Возможные значения: `720p`, `1080p`, `4k`
</ParamField>

Сгенерировано из схемы, которую Router отдаёт по `GET /v2/models/kling/kling-4.0/openapi.json`; это тот же документ, по которому он проверяет вызов, прежде чем запрос достигнет провайдера.
