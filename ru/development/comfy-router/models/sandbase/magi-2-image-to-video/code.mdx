```python Python
import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(

                "duration": 5,

            },
        )

        # Опрашивать до завершения запроса, выдерживая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, здесь вызывает типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("sandbase/magi-2-image-to-video", {
  aspect_ratio: "16:9",
  duration: 5,
  generate_audio: true,
  image: "https://cdn.example.com/first-frame.jpg",
  prompt: "The jazz band starts playing; the camera slowly pushes in.",
  resolution: "1080p",
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивать до завершения запроса, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, завершившийся ошибкой или отменённый, отклоняется здесь.
const result = await handle.get();

console.log(result.data);
```
