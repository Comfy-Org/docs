import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.
# Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "tencent/hunyuan-image-3.5",
            {
                "aspect_ratio": "16:9",
                "output_format": "png",
                "prompt": "a serene mountain lake at sunrise, ultra detailed, cinematic lighting",

            },
        )

        # Опрашивайте, пока запрос не завершится, выжидая Retry-After, указанный сервером.
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # Собственный payload провайдера, то же значение, что возвращает models.run().
        # Запрос, который завершился ошибкой или был отменён, вызывает здесь типизированную ошибку Router.
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("tencent/hunyuan-image-3.5", {
  aspect_ratio: "16:9",
  output_format: "png",
  prompt: "a serene mountain lake at sunrise, ultra detailed, cinematic lighting",
  resolution: "1080p",
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выжидая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.
const result = await handle.get();

console.log(result.data);

  Референсные изображения, по публичному https URL. На этом маршруте ОТКЛОНЯЕТСЯ: Hunyuan Image 3.5 запущен только в текстовом режиме, поэтому непустой `images` отклоняется с ошибкой 400. MPS загружает референсное изображение только по URL, доступному краулеру, и не может принимать очищенные байты, поэтому данные о местоположении из EXIF в изображениях вызывающего нельзя удалить заранее; ввод изображений пока не поддерживается. Отправляйте только промпт.
</ParamField>

  Возможные значения: `1`
