import asyncio
from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменных окружения.

async def main():

        # Собственный payload провайдера, то же значение, что возвращает models.run().

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменных окружения.
// Каждый вызов submit() создаёт собственный Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("tencent/hunyuan-3d-texture-edit", {

    Type: "FBX",
    Url: "https://vcg-test-1258344699.cos.ap-guangzhou.myqcloud.com/test/3d/test.fbx",
  },
  Prompt: "a kitten",
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивайте, пока запрос не завершится, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, который завершился ошибкой или был отменён, отклоняется здесь.
const result = await handle.get();

console.log(result.data);

<ResponseField name="Response.ResultFile3Ds[].Type" type="string">
  Формат файла. Завершённая задача может вернуть несколько записей
в произвольном порядке, и не каждая запись является сеткой:
замечено, что операции UV и smart-topology возвращают сетки OBJ,
GLB и FBX вместе с IMAGE (текстурой PNG). Выбирайте запись по
`Type`, а не по положению. Это намеренно не enum: приведённые
значения лишь те, что наблюдались, этот компонент используется
совместно с операциями, вывод которых никогда не наблюдался
(part, texture-edit), а схема вывода носит описательный характер,
поэтому значение, которого нет в списке и которое возвращает
Tencent, не должно превращать корректный результат в нарушение
схемы.
