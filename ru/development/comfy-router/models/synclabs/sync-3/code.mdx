from comfy_sdk import AsyncComfy

# Читает COMFY_API_KEY из переменной окружения.

        # Собственный ответ провайдера, то же значение, что возвращает models.run().
        # Запрос, завершившийся ошибкой или отменённый, вызовет здесь типизированную ошибку Router.
        result = await handle.get()

import { comfy } from "@comfyorg/sdk";

// Читает COMFY_API_KEY из переменной окружения.
// Каждый вызов submit() создаёт свой Idempotency-Key и переиспользует его при автоматических повторах.
const handle = await comfy.models.submit("synclabs/sync-3", {
  input: [
    {
      type: "video",
      url: "https://example.invalid/synclabs/sync-3/speaker.mp4",
    },
    {
      type: "audio",
      url: "https://example.invalid/synclabs/sync-3/voiceover.wav",
    },
  ],
  options: {
    sync_mode: "bounce",
  },
});
console.log("requestId:", handle.requestId); // вместе с ID модели это всё, что нужно другому процессу

// Опрашивать до завершения запроса, выдерживая Retry-After, указанный сервером.
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// Тот же результат, что возвращает models.run(). Запрос, завершившийся ошибкой или отменённый, будет отклонён здесь.
const result = await handle.get();

console.log(result.data);
