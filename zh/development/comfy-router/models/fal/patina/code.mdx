import Foundation
import ComfySwiftSDK

// Reads COMFY_API_KEY from the environment.
// Each submit() call mints its own Idempotency-Key and reuses it for automatic retries.
let client = ComfyCloudClient(apiKey: ProcessInfo.processInfo.environment["COMFY_API_KEY"]!)

    "fal/patina",
    input: [
        "image_url": "https://storage.googleapis.com/falserverless/gallery/patina-blog-hero-render.png",
    ]

print("requestId:", handle.requestId)  // with the model ID, all another process needs

// Poll until the request completes, waiting the Retry-After the server names.
for try await update in handle.events() {
    print(update.state.rawValue, update.queuePosition.map(String.init) ?? "unknown")
}

// The provider's own payload, the same value models.run() returns.
// A request that failed or was cancelled throws the typed Router error here.
let result = try await handle.result()

# 1. Submit. Router answers 201 with request_id, status_url, response_url and cancel_url.

  -H "X-API-Key: $COMFY_API_KEY" \

  -d "{\"image_url\": \"https://storage.googleapis.com/falserverless/gallery/patina-blog-hero-render.png\"}"

  -H "X-API-Key: $COMFY_API_KEY"

# 3. Collect. 200 with the model's native output, 202 with the status body while it is still running.
curl https://api.comfy.org/v2/models/fal/patina/requests/$REQUEST_ID \

## Schema
### 输入

<ParamField body="enable_safety_checker" type="boolean" default="true">
  为图像启用安全检查器。
</ParamField>

<ParamField body="image_url" type="string" required>
  输入图像的 URL（照片或渲染图）。
</ParamField>

<ParamField body="maps" type="string[]" default="[&quot;basecolor&quot;,&quot;normal&quot;,&quot;roughness&quot;,&quot;metalness&quot;,&quot;height&quot;]">
  要预测哪些 PBR 贴图。默认为全部五种。
</ParamField>

<ParamField body="output_format" type="string" default="&quot;png&quot;">
  输出图像格式：jpeg、png 或 webp。
</ParamField>

<ParamField body="seed" type="integer">
  用于可复现降噪的随机种子。
</ParamField>

<ParamField body="sync_mode" type="boolean" default="false">
  如果为 true，则以 data URI 形式返回图像，而不是 CDN URL。
</ParamField>

该模式由 Router 在 `GET /v2/models/fal/patina/openapi.json` 提供，也是在请求到达提供商之前用于校验调用的同一份文档。

### 输出

<ResponseField name="images" type="object[]">
  已生成的可平铺纹理图像，以及预测得到的 PBR 材质贴图。基础纹理只有 url；每个贴图条目还带有 map_type。
</ResponseField>

<ResponseField name="images[].map_type" type="string">
  PBR 贴图类型（basecolor、normal、roughness、metalness、height）。基础纹理图像没有此项。
</ResponseField>

<ResponseField name="images[].url" type="string">
  已生成图像的 URL。
</ResponseField>

<ResponseField name="prompt" type="string">
  用于纹理生成的提示词（可能经过扩展）。
</ResponseField>

<ResponseField name="seed" type="integer">
  用于纹理生成的种子。
</ResponseField>

<ResponseField name="timings" type="object">
  端到端耗时明细（秒）。
</ResponseField>
