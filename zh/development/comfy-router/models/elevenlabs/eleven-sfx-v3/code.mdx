import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "elevenlabs/eleven_sfx_v3",
            {
                "duration_seconds": 5,
                "text": "A distant rumble of thunder rolling across a valley.",
            },
        )
        print("request_id:", handle.request_id)  # 配合模型 ID，这就是另一个进程所需的全部信息

        # 轮询直到请求完成，并按服务器指定的 Retry-After 等待。

        # 提供商自身的负载，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在此处抛出类型化的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

}

const result = await handle.get();

必须至少为 0.5，至多为 30。
此路由上为必填：与上游 ElevenLabs API 不同（该 API 在字段为 null 时会猜测一个最佳时长），此路由会以 400 "Duration is required" 拒绝省略该字段的请求。
该字段保持可为空，只是为了让显式 null 成为格式良好的文档；但它仍会被拒绝。
请求按此值计量。

  范围：`0.5` 到 `30`

  格式：`double`
</ParamField>

<ParamField body="loop" type="boolean" default="false">
  是否创建可平滑循环的音效。
</ParamField>

<ParamField body="model_id" type="string">
  用于声音生成的模型 ID。此路由仅接受 v1 请求枚举中的 Comfy ID；任何其他值都会在请求到达 ElevenLabs 之前被拒绝并返回 400。每个 ID 都是 Comfy 为 ElevenLabs 模型使用的 ID（'eleven_sfx_v2' 是 ElevenLabs 称为 'eleven_text_to_sound_v2' 的模型）。代理会将转发的请求体重写为供应商使用的拼写，这是 ElevenLabs 自己的枚举所要求的（它会以 422 拒绝 'eleven_sfx_v2'），因此调用方永远不会发送供应商 ID。它并不在此 schema 的 `required` 列表中，因为 Comfy Router 会从 /v2/models/elevenlabs/\{model} 的 `{model}` 路径段填充它，所以 Router 调用方会省略它。
</ParamField>

  范围：`0` 到 `1`
