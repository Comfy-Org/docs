import { comfy } from "@comfyorg/sdk";

// 从环境中读取 COMFY_API_KEY。
// 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。

// 轮询直到请求完成，每次等待服务器指定的 Retry-After。

}

// 与 models.run() 返回的结果相同。失败或已取消的请求会在此处被拒绝。
const result = await handle.get();
