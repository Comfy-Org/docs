</ParamField>

  来源图像的 Comfy 签名资产 URL。

  格式：`uri`
</ParamField>

<ParamField body="remesh" type="string">

  可能值：`none`、`triangle`、`quad`

  可能值：`none`、`vertex`、`face`
</ParamField>
