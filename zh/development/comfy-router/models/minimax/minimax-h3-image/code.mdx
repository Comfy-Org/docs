```python Python
import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次调用 submit() 都会生成自己的 Idempotency-Key，并在自动重试时复用。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "minimax/minimax-h3-image",
            {
                "area": "1K",
                "prompt": "A single red maple leaf resting on a plain white background.",
            },
        )
        print("request_id:", handle.request_id)  # 配合模型 ID，这就是另一个进程所需的全部信息

        # 轮询直到请求完成，等待服务器指定的 Retry-After。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # 提供商自己的载荷，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在这里抛出类型化的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// 从环境变量读取 COMFY_API_KEY。
// 每次调用 submit() 都会生成自己的 Idempotency-Key，并在自动重试时复用。
const handle = await comfy.models.submit("minimax/minimax-h3-image", {
  area: "1K",
  prompt: "A single red maple leaf resting on a plain white background.",
});
console.log("requestId:", handle.requestId); // 配合模型 ID，这就是另一个进程所需的全部信息

// 轮询直到请求完成，等待服务器指定的 Retry-After。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// 与 models.run() 返回的结果相同。失败或已取消的请求会在这里被拒绝。
const result = await handle.get();

console.log(result.data);
```

<ParamField body="area" type="string">
  当省略 `size` 或 `size` 为 `auto` 时的输出面积级别。取值为 `1K`、`2K`、`4K`（默认）。
</ParamField>

<ParamField body="aspect_ratio" type="string">
  当省略 `size` 或 `size` 为 `auto` 时的输出宽高比。`auto`（默认），或 1:3 到 3:1 之间的 `X:Y`，例如 `16:9`。
</ParamField>

<ParamField body="disable_prompt_enhanced" type="boolean" default="false">
  关闭 MiniMax 的提示词增强。需要搭配固定的 `size` 或 `X:Y` 形式的 `aspect_ratio`。
</ParamField>

<ParamField body="images" type="object[]">
  可选的参考图像。存在时，请求将以图像编辑方式执行，与编辑路由完全一致。
</ParamField>

<ParamField body="images[].image_url" type="string" required>
  公开的 http(s) URL，或 `data:image/<format>;base64,` URI。支持 JPEG、PNG、WEBP、HEIC 或 HEIF，每边 128 至 8192 px，长边不超过短边的 4 倍。
</ParamField>

<ParamField body="model" type="string">
  模型名。`MiniMax-H3-Image` 或 `MiniMax-H3-Image-Highspeed`。Router 调用方可以省略此字段或发送 null；Router 会填入由请求路径所选的模型。
</ParamField>

<ParamField body="n" type="integer" default="1">
  图像数量。输入 token 按每张图像计费一次。

  范围：`1` 到 `10`
</ParamField>

<ParamField body="output_compression" type="integer">
  `jpeg` 和 `webp` 的压缩质量，默认 95。

  范围：`0` 到 `100`
</ParamField>

<ParamField body="output_format" type="string">
  输出图像格式。取值为 `png`、`jpeg`、`jpg`（默认）、`webp`。
</ParamField>

<ParamField body="prompt" type="string" required>
  图像的文字描述。与 `images` 一起使用时，为编辑的描述；参考图像按数组顺序依次为图像 1、图像 2，以此类推。
</ParamField>

<ParamField body="size" type="string">
  输出尺寸，`auto`（默认）或 `WIDTHxHEIGHT`，例如 `2720x1536`。固定尺寸的面积必须为 655,360 至 20,000,000 像素，宽高比在 1:3 到 3:1 之间，且不能与 `aspect_ratio` 或 `area` 同时使用。
</ParamField>
