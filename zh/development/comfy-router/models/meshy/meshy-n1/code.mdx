        # 轮询直到请求完成，每次都等待服务器指定的 Retry-After 时间。

import { comfy } from "@comfyorg/sdk";

// 从环境变量中读取 COMFY_API_KEY。
// 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。

  image_style_url: "https://example.invalid/meshy/n1/front.png",

console.log("requestId:", handle.requestId); // 配合模型 ID，就是另一个进程所需的全部信息

// 轮询直到请求完成，每次都等待服务器指定的 Retry-After 时间。
for await (const update of handle.events()) {

}

const result = await handle.get();

<ParamField body="model_url" type="string">
  指向 3D 模型的公开可访问 URL 或 data URI。支持的格式为 glb、gltf、obj、fbx、stl。如果未提供 input_task_id，则此项必填。最多 7,999,999 个三角面。

<ParamField body="target_formats" type="string[]">

</ParamField>

<ParamField body="texture_resolution" type="string" default="&quot;2k&quot;">
  基础颜色贴图的分辨率。meshy-n1 仅支持 2k。

  可能的值：`2k`
</ParamField>
