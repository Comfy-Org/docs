```python Python
import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量中读取 COMFY_API_KEY。
# 每次调用 submit() 都会生成自己的 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "meshy/meshy-6",
            {
                "art_style": "realistic",
                "mode": "preview",
                "prompt": "a red cube",
            },
        )
        print("request_id:", handle.request_id)  # 配合模型 ID，这就是另一个进程所需的全部信息

        # 轮询直到请求完成，按服务器给出的 Retry-After 等待。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # 提供商自身的响应负载，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在这里抛出带类型的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// 从环境变量中读取 COMFY_API_KEY。
// 每次调用 submit() 都会生成自己的 Idempotency-Key，并在自动重试时复用它。
const handle = await comfy.models.submit("meshy/meshy-6", {
  art_style: "realistic",
  mode: "preview",
  prompt: "a red cube",
});
console.log("requestId:", handle.requestId); // 配合模型 ID，这就是另一个进程所需的全部信息

// 轮询直到请求完成，按服务器给出的 Retry-After 等待。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// 与 models.run() 返回的结果相同。失败或已取消的请求会在这里被拒绝。
const result = await handle.get();

console.log(result.data);
```

```bash cURL
# 1. 提交。Router 返回 201，并带有 request_id、status_url、response_url 和 cancel_url。
curl https://api.comfy.org/v2/models/meshy/meshy-6/requests \
  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d "{\"art_style\": \"realistic\", \"mode\": \"preview\", \"prompt\": \"a red cube\"}"

# 2. 轮询直到状态为 COMPLETED，按每个响应给出的 Retry-After 秒数等待。
REQUEST_ID="<request_id from the 201 body>"
curl -i https://api.comfy.org/v2/models/meshy/meshy-6/requests/$REQUEST_ID/status \
  -H "X-API-Key: $COMFY_API_KEY"

# 3. 收集。就绪时返回 200 和模型的原始输出，仍在运行时返回 202 和状态响应体。
curl https://api.comfy.org/v2/models/meshy/meshy-6/requests/$REQUEST_ID \
  -H "X-API-Key: $COMFY_API_KEY"
```

## Schema

### 输入

#### `preview` 变体

<ParamField body="ai_model" type="string">
  要运行的 Meshy 模型。在 Comfy Router 路由上，该值由路径（`/v2/models/meshy/{model}`）提供，可以省略或发送为 null；验证之后，Router 会将路径中的模型写入此处。在 v1 /proxy 路由上参见 MeshyTextTo3DPreviewRequest，它将其约束为 MeshyAiModel。
</ParamField>

<ParamField body="art_style" type="string" default="&quot;realistic&quot;">
  描述你期望的对象艺术风格。

  可选值：`realistic`、`sculpture`
</ParamField>

<ParamField body="geometry_resolution" type="string" default="&quot;standard&quot;">
  已生成网格的几何分辨率。2k 和 4k 会运行 Ultra 通道，以获得更精细的表面细节。2k 和 4k 要求 ai_model 为 meshy-7.1 或最新。

  可选值：`standard`、`2k`、`4k`
</ParamField>

<ParamField body="is_a_t_pose" type="boolean" default="false">
  弃用。请改用 pose_mode。是否以 A/T 姿势生成模型。
</ParamField>

<ParamField body="mode" type="string" required>
  创建预览任务时，该字段应设置为 "preview"。

  可选值：`preview`
</ParamField>

<ParamField body="moderation" type="boolean" default="false">
  当为是时，输入内容将被筛查是否存在潜在有害内容。
</ParamField>

<ParamField body="pose_mode" type="string">
  指定已生成模型的姿势模式。

  可选值：`a-pose`、`t-pose`、``
</ParamField>

<ParamField body="prompt" type="string" required>
  描述该 3D模型 是什么类型的对象。最多 800 个字符。
</ParamField>

<ParamField body="should_remesh" type="boolean" default="false">
  控制是否启用重网格化阶段。当为否时，返回最高精度的三角网格。Meshy 5 默认为是，Meshy 6 和 7 默认为否。
</ParamField>

<ParamField body="symmetry_mode" type="string" default="&quot;auto&quot;">
  控制模型生成过程中的对称行为。

  可选值：`off`、`auto`、`on`
</ParamField>

<ParamField body="target_polycount" type="integer" default="30000">
  指定已生成模型的目标多边形数量。有效范围是 100 到 300,000。

  范围：`100` 到 `300000`
</ParamField>

<ParamField body="topology" type="string" default="&quot;triangle&quot;">
  指定已生成模型的拓扑。

  可选值：`quad`、`triangle`
</ParamField>

<ParamField body="ultra_mode" type="boolean" default="false">
  弃用。请改用 geometry_resolution；取值为是时等价于 geometry_resolution 2k。仅当 ai_model 为 meshy-7、meshy-7.1 或最新时才受支持。meshy-7 是例外：geometry_resolution 2k 要求 meshy-7.1 或最新，因此在 meshy-7 上，要请求 Ultra 通道需将 ultra_mode 设为是。
</ParamField>
