<ResponseField name="duration" type="number">
  生成视频的时长，单位为秒。声明为 number 而非 integer，是因为 BytePlus 在这一点上并不一致：已观察到视频任务返回整数秒，而同属 BytePlus 的其他接口会报告小数时长，因此客户端不能假定它是整数值。这是 BytePlus 自己的字段，在成功的视频任务上返回并原样转发。
</ResponseField>

<ResponseField name="error" type="object">

</ResponseField>

<ResponseField name="error.code" type="string">

</ResponseField>

<ResponseField name="error.message" type="string">
  报错信息
</ResponseField>

<ResponseField name="id" type="string"><ResponseField name="id" type="string">
  视频生成任务的 ID
</ResponseField>

<ResponseField name="model" type="string">
  该任务所使用的模型名称和版本
</ResponseField>

<ResponseField name="output_format" type="string">
  生成视频的容器格式（mp4 或 mov），在顶层作为 `content` 的同级字段返回，Seedance 视频任务查询即在此处返回该字段。这是 BytePlus 自身的字段，原样透传。
</ResponseField>

<ResponseField name="resolution" type="string">
  生成视频的分辨率，例如 `1080p`。这是 BytePlus 自身的字段，在成功的视频任务中返回，原样透传。
</ResponseField>

<ResponseField name="seed" type="integer">

</ResponseField>

<ResponseField name="status" type="string">

<ResponseField name="updated_at" type="integer">
