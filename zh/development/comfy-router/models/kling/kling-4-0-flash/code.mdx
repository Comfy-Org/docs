```python Python
import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "kling/kling-4.0-flash",
            {
                "prompt": "A neon-lit alley in the rain, slow dolly forward.",
                "settings": {
                    "aspect_ratio": "16:9",
                    "duration": 5,
                    "resolution": "720p",
                },
            },
        )
        print("request_id:", handle.request_id)  # 与模型 ID 一起，就是另一个进程所需的全部

        # 轮询直到请求完成，等待服务器指定的 Retry-After。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # 提供商自身的 payload，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在此处抛出带类型的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())
```

```typescript TypeScript
import { comfy } from "@comfyorg/sdk";

// 从环境变量读取 COMFY_API_KEY。
// 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。
const handle = await comfy.models.submit("kling/kling-4.0-flash", {
  prompt: "A neon-lit alley in the rain, slow dolly forward.",
  settings: {
    aspect_ratio: "16:9",
    duration: 5,
    resolution: "720p",
  },
});
console.log("requestId:", handle.requestId); // 与模型 ID 一起，就是另一个进程所需的全部

// 轮询直到请求完成，等待服务器指定的 Retry-After。
for await (const update of handle.events()) {
  console.log(update.status, update.queuePosition);
}

// 与 models.run() 返回的结果相同。失败或已取消的请求会在此处拒绝。
const result = await handle.get();

console.log(result.data);
```

# 1. 提交。Router 返回 201，并带有 request_id、status_url、response_url 和 cancel_url。

  -H "X-API-Key: $COMFY_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \

# 2. 轮询直到状态为 COMPLETED，每次响应会指定需要等待的 Retry-After 秒数。

  -H "X-API-Key: $COMFY_API_KEY"

# 3. 收集。返回 200 时为模型的原生输出；仍在运行时返回 202 和状态响应体。

## Schema

### 输入

<ParamField body="options" type="object">
  常规配置，例如水印选项。Kling 4.0 不支持 `options.callback_url` 和 `options.external_task_id`；Comfy Router 会拒绝在 `options` 中设置其中任一字段的请求。顶层的 `callback_url` 或 `external_task_id` 不属于此 schema，并会在请求到达 Kling 之前被移除。
</ParamField>

<ParamField body="options.watermark_info" type="object">
  是否同时生成带水印的结果。不支持自定义水印。显式的 null 会被视为不存在，这与 /proxy 的 Kling 4.0 检查和 Kling 本身的行为一致。
</ParamField>

<ParamField body="options.watermark_info.enabled" type="boolean">
  true 表示生成带水印的结果，false 表示不生成。默认 false。
</ParamField>

<ParamField body="prompt" type="string" required>
  可包含正面和负面描述的提示词。最多 26,000 个英文字符或 11,000 个非英文字符；comfy-api 强制执行 1 到 26,000 个字符的限制，而 Kling 本身会拒绝更长的非英文文本。
</ParamField>

<ParamField body="settings" type="object" required>
  输出配置，例如分辨率、宽高比和时长。
</ParamField>

<ParamField body="settings.aspect_ratio" type="string">
  已生成帧的宽高比（宽度:高度）。取值为 "21:9"、"16:9"、"9:16"、"1:1"、"4:3" 或 "3:4" 之一。默认 "16:9"。

  可能的值：`21:9`、`16:9`、`9:16`、`1:1`、`4:3`、`3:4`
</ParamField>

<ParamField body="settings.duration" type="integer" required>
  视频时长，以秒为单位，从 3 到 20。必填。

  范围：`3` 到 `20`
</ParamField>

<ParamField body="settings.resolution" type="string" required>
  已生成视频的分辨率。Kling 4.0 Flash 仅渲染 "720p"。必填。

  可能的值：`720p`
</ParamField>
