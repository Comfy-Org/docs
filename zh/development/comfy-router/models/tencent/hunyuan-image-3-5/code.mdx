import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "tencent/hunyuan-image-3.5",
            {
                "aspect_ratio": "16:9",
                "output_format": "png",
                "prompt": "a serene mountain lake at sunrise, ultra detailed, cinematic lighting",
                "resolution": "1080p",
            },
        )
        print("request_id:", handle.request_id)  # 配合模型 ID，就是另一个进程所需的全部信息

        # 轮询直到请求完成，等待时长按服务器给出的 Retry-After 为准。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # 提供商自身的响应数据，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在此处抛出类型化的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())

## Schema

### 输入

<ParamField body="aspect_ratio" type="string">
  宽高比，以 W:H 形式表示，例如 16:9。
</ParamField>

<ParamField body="images" type="string (uri)[]">
  参考图像，通过公开的 https URL 提供。此路由会拒绝该参数：混元图像 3.5 以纯文本模式发布，因此非空的 `images` 会以 400 被拒绝。MPS 只能通过爬虫可访问的 URL 获取参考图像，无法接收已剥离元数据的字节流，因此调用方的图像无法事先移除其 EXIF 位置数据；目前尚不支持图像输入。请仅发送 prompt。
</ParamField>

<ParamField body="n" type="integer" default="1">
  要生成的图像数量。仅提供 1 张（即计费数量）。

  可能的值：`1`
</ParamField>

<ParamField body="negative_prompt" type="string">
  描述图像应避免出现的内容的文本。
</ParamField>

<ParamField body="output_format" type="string">
  生成输出的图像编码格式。可选 png 或 jpeg。

  可能的值：`png`、`jpeg`
</ParamField>

<ParamField body="prompt" type="string" required>
  描述要生成图像的文本 prompt。必填：此路由仅支持文本，因此 prompt 是唯一被接受的输入。签名之前会先对其进行裁剪。
</ParamField>

<ParamField body="resolution" type="string" default="&quot;1080p&quot;">
  输出分辨率档位，同时也是计费档位。可选 720p、1080p、2k 或 4k。

  可能的值：`720p`、`1080p`、`2k`、`4k`
</ParamField>
