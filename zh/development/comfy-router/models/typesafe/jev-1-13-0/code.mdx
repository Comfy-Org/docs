import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次 submit() 调用都会自行生成 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "typesafe/jev-1.13.0",
            {
                "questions": {
                    "department": {
                        "criteria": {
                            "billing": "Payments, invoicing, refunds",
                            "technical": "Bugs, outages, integrations",
                        },

                        "type": "choice",
                    },
                    "is_urgent": {
                        "instructions": "Does this convey urgency?",
                        "type": "noul",
                    },
                },
                "state": "Help! My payouts have been failing for 3 days.",
            },
        )
        print("request_id:", handle.request_id)  # 加上模型 ID，就是另一个进程所需的全部信息

        # 轮询直到请求完成，按服务器给出的 Retry-After 等待。
        async for update in handle.iter_events():
            print(update.status, update.queue_position)

        # 提供商自己的载荷，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在此处抛出带类型的 Router 错误。

    print(result)

asyncio.run(main())

import { comfy } from "@comfyorg/sdk";

// 从环境变量读取 COMFY_API_KEY。
// 每次 submit() 调用都会自行生成 Idempotency-Key，并在自动重试时复用它。

      criteria: {
        billing: "Payments, invoicing, refunds",

      },
      instructions: "Which team should handle this?",
      type: "choice",
    },
    is_urgent: {
      instructions: "Does this convey urgency?",
      type: "noul",
    },
  },
  state: "Help! My payouts have been failing for 3 days.",
});
console.log("requestId:", handle.requestId); // 加上模型 ID，就是另一个进程所需的全部信息

// 轮询直到请求完成，按服务器给出的 Retry-After 等待。

## Schema

### 输入

<ParamField body="assets" type="object">
  请求中的图像，以引用字符串为键。每个引用都必须出现在 `state`、`instructions` 或 `criteria` 中的某处。每张图像都是静态 PNG 或 JPEG，每边 280 到 1568 像素，解码后最大 16 MiB；TypeSafe 会拒绝总计超过 32 MiB 的调用。
</ParamField>

<ParamField body="model" type="string">
  固定的 Jev 版本。可选：Router 会根据路径中的模型 ID 填充它。不接受 `jev-latest` 和 `jev-preview` 别名。
</ParamField>

<ParamField body="questions" type="object" required>
  要回答的问题，以你选择的 ID 为键。每个问题的答案都会在相同的 ID 下返回。
</ParamField>

<ParamField body="state" type="object" required>
  要评估的内容：一个字符串，或最深达 64 层的结构化数据（对象或数组）。其中任何位置的资产引用都会被替换为对应的图像。
</ParamField>

### 输出

<ResponseField name="answers" type="object" required>
