import asyncio
from comfy_sdk import AsyncComfy

# 从环境变量读取 COMFY_API_KEY。
# 每次 submit() 调用都会生成自己的 Idempotency-Key，并在自动重试时复用它。
async def main():
    async with AsyncComfy() as client:
        handle = await client.models.submit(
            "typesafe/jev-1.14.0",
            {
                "questions": {
                    "department": {
                        "criteria": {

                        "type": "noul",

                },
                "state": "Help! My payouts have been failing for 3 days.",
            },
        )
        print("request_id:", handle.request_id)  # 连同模型 ID，这是另一个进程所需的全部信息

        # 轮询直到请求完成，等待服务器指定的 Retry-After。
        async for update in handle.iter_events():

        # 提供商自己的负载，与 models.run() 返回的值相同。
        # 失败或已取消的请求会在这里抛出类型化的 Router 错误。
        result = await handle.get()

    print(result)

asyncio.run(main())

    },
    is_urgent: {
      instructions: "Does this convey urgency?",
      type: "noul",

});
console.log("requestId:", handle.requestId); // 连同模型 ID，这是另一个进程所需的全部信息

// 轮询直到请求完成，等待服务器指定的 Retry-After。

## Schema

### 输入

<ParamField body="assets" type="object">
  请求中的图像，以引用字符串为键。每个引用都必须出现在 `state`、`instructions` 或 `criteria` 中的某处。每张图像都是静帧 PNG 或 JPEG，每边 280 至 1568 像素，解码后至多 16 MiB；TypeSafe 会拒绝总计超过 32 MiB 的调用。
</ParamField>

<ParamField body="model" type="string">
  固定的 Jev 版本。可选：Router 会根据路径中的模型 id 填充它。不接受 `jev-latest` 和 `jev-preview` 别名。
</ParamField>

<ParamField body="questions" type="object" required>
  需要回答的问题，以你选择的 id 为键。每个问题的答案都会在相同的 id 下返回。
</ParamField>

<ParamField body="state" type="object" required>
  待评估的内容：字符串，或最多 64 层深的结构化数据（对象或数组）。其中任何位置的资产引用都会被替换为该图像。
</ParamField>

根据 Router 在 `GET /v2/models/typesafe/jev-1.14.0/openapi.json` 提供的 schema 生成，这也是它在请求到达提供商之前用于验证调用的同一份文档。

### 输出

<ResponseField name="answers" type="object" required>

<ResponseField name="model" type="string" required>

</ResponseField>
